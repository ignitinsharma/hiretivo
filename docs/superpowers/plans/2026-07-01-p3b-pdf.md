# Phase 3b — ATS-Safe PDF Resume Renderer Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking. One commit per task.

**Goal:** Install `@react-pdf/renderer`, define a typed `ResumeDocument` model, and build a pure render function (`src/lib/resume/render.ts`) that produces an **ATS-safe PDF** (single-column, standard system fonts, real selectable text) from seeker profile data and uploads it to R2 at key `resumes/{seekerId}/...`. This is the artifact the Phase 5 apply wizard consumes: Phase 4 (AI) tailors a `ResumeDocument`, Phase 5 calls `renderResumePdf()` to materialise + store the PDF, then deep-links the ATS.

**Architecture:** `render.ts` is a **pure, framework-agnostic library** — no Next.js request context, no Prisma calls, no `auth()`. It accepts an already-assembled `ResumeDocument` (typed) and returns a `Buffer`; a thin sibling (`store.ts`) uploads that buffer to R2 via the existing `uploadFile` + `R2_KEYS.resumeGenerated` key helper. Data assembly (Prisma → `ResumeDocument`) lives in a separate mapper so callers (Phase 4 AI tailoring, Phase 5 wizard, this phase's smoke script) all build the same typed input the same way. The `ResumeDocument` type is the stable contract Phase 4 writes to and Phase 5 reads from. Certifications and projects are accepted as plain typed arrays on `ResumeDocument` (not read from Prisma here), so 3b is NOT blocked by whether Phase 3a's `Certification` model has landed — the mapper fills them from whatever source exists at call time.

**Tech Stack:** Next.js 14 App Router, TypeScript (strict), `@react-pdf/renderer` (React-based declarative PDF), Prisma (read-only, in the mapper), Cloudflare R2 via existing `src/lib/r2.ts`, `date-fns`, `tsx` for the smoke script, `node --test` for unit tests.

- **Types:** never `any`. Enums (in `src/constants/enums.ts`), never string-literal unions. Path aliases (`@/…`), never `../../../`.

### ATS-Safe Acceptance Criteria (bake into every template task)

- [ ] **Single column only.** No multi-column `flexDirection: 'row'` for content flow. Row layout permitted ONLY for inline header meta (name row, contact line) and a skills chip line — never to split the body into parallel columns an ATS would read out of order.
- [ ] **Standard fonts only.** Use react-pdf built-in `Helvetica` / `Helvetica-Bold` / `Times-Roman` (or `Courier`). **Do NOT register custom/remote fonts** (no `Font.register` with URLs — violates CSP-style offline constraint and risks non-embedded glyphs). System-standard fonts guarantee real embedded text.
- [ ] **Real selectable text.** Every piece of content is a `<Text>`. No content rendered as `<Image>`, no icon fonts for content, no SVG-as-text. Decorative rules may be `<View>` borders only.
- [ ] **No tables-for-content.** No `<View>` grid emulating a table for experience/education. Linear stacked sections only.
- [ ] **Standard section headings.** Exact, ATS-recognised, uppercase-or-titlecase words: `SUMMARY`, `SKILLS`, `EXPERIENCE`, `EDUCATION`, `CERTIFICATIONS`, `PROJECTS`. These live in a `ResumeSectionHeading` enum — never inline strings.
- [ ] **Reading order = DOM order.** Sections render top-to-bottom in the fixed order above; skills grouped by category but each group is a single wrapping `<Text>` line, not a grid.
- [ ] **Selectable-text proof** recorded in the smoke step (copy-paste check above).

```ts
import type { ResumeSectionHeading } from "@/constants/enums";

/*
 * ResumeContact
 * Header contact line. Optional fields render only when present (ATS-safe:
 * missing fields are omitted, never rendered as empty labels).
 */
export interface ResumeContact {
  email?: string;
  location?: string;
  websiteUrl?: string;
  linkedinUrl?: string;
  githubUrl?: string;
}

/*
 * ResumeSkillGroup
 * One skill category ("Engineering", "Data", …) and its flat skill names.
 * Rendered as a single wrapping <Text> line per group (never a grid).
 */
export interface ResumeSkillGroup {
  category: string;
  skills: string[];
}

/*
 * ResumeExperience
 * One work-history entry. Dates are pre-formatted display strings
 * ("Jan 2023"), never Date objects — freeform "MM/YYYY" origin.
 */
export interface ResumeExperience {
  company: string;
  title: string;
  startDate: string;
  endDate: string | null; // null => "Present"
  isCurrent: boolean;
  bullets: string[]; // description split into ATS-friendly bullet lines
}

/*
 * ResumeEducation
 * One education entry. All optional except institution.
 */
export interface ResumeEducation {
  institution: string;
  degree?: string;
  major?: string;
  graduationDate?: string;
}

/*
 * ResumeCertification
 * One certification. Supplied by the caller (Phase 3a Certification rows or
 * parsed data) — not read from Prisma inside the renderer.
 */
export interface ResumeCertification {
  name: string;
  issuer?: string;
  issueDate?: string;
}

/*
 * ResumeProject
 * One portfolio project. Maps from SeekerProfile.topProjects Json
 * ({ title?, url?, description? }).
 */
export interface ResumeProject {
  title: string;
  url?: string;
  description?: string;
}

/*
 * ResumeDocument
 * The complete, render-ready resume model. This is the Phase 3b <-> Phase 4
 * <-> Phase 5 contract: Phase 4 AI produces/tailors this; Phase 5 passes it to
 * renderResumePdf(). Fully typed, no `any`, no Prisma types leaking through.
 * Sections render in RESUME_SECTION_ORDER; a section is skipped entirely when
 * its array is empty (or summary is blank).
 */
export interface ResumeDocument {
  fullName: string;
  headline?: string; // e.g. "Senior Backend Engineer"
  contact: ResumeContact;
  summary?: string;
  skillGroups: ResumeSkillGroup[];
  experience: ResumeExperience[];
  education: ResumeEducation[];
  certifications: ResumeCertification[];
  projects: ResumeProject[];
}

/*
 * ResumeSectionKey
 * Re-export alias so template code can key section renderers by heading.
 */
export type ResumeSectionKey = ResumeSectionHeading;
```

- [ ] **Step 2: Section sub-components**

`src/lib/resume/template/sections.tsx`. Each is a small pure component (<80 lines) that renders one section as a single-column stack of `<Text>` nodes. An empty section renders `null` (caller decides order/skip). Note the `entryHeaderRow` row usage is header-meta only (title left, dates right) — permitted by the ATS criteria; the body stays single-column.

```tsx
import { Text, View, Link } from "@react-pdf/renderer";
import { ResumeSectionHeading } from "@/constants/enums";
import { resumeStyles as s } from "@/lib/resume/template/styles";
import type {
  ResumeSkillGroup,
  ResumeExperience,
  ResumeEducation,
  ResumeCertification,
  ResumeProject,
} from "@/modules/seeker/profile/types/resume-document.types";

/*
 * SectionHeading
 * Renders one standard ATS heading. Pulls text from the enum so headings are
 * never free strings.
 * @param heading - one ResumeSectionHeading value
 */
export function SectionHeading({ heading }: { heading: ResumeSectionHeading }) {
  return <Text style={s.sectionHeading}>{heading}</Text>;
}

/*
 * SummarySection
 * Single paragraph. Renders null when summary is blank (section skipped).
 * @param summary - optional summary text
 */
export function SummarySection({ summary }: { summary?: string }) {
  if (!summary?.trim()) return null;
  return (
    <View>
      <SectionHeading heading={ResumeSectionHeading.SUMMARY} />
      <Text style={s.paragraph}>{summary}</Text>
    </View>
  );
}

/*
 * SkillsSection
 * One wrapping <Text> line per category ("Engineering: React, Node.js").
 * Never a grid — keeps ATS reading order linear.
 * @param groups - skill groups by category
 */
export function SkillsSection({ groups }: { groups: ResumeSkillGroup[] }) {
  if (!groups.length) return null;
  return (
    <View>
      <SectionHeading heading={ResumeSectionHeading.SKILLS} />
      {groups.map((g) => (
        <Text key={g.category} style={s.skillGroupLine}>
          <Text style={s.skillGroupLabel}>{g.category}: </Text>
          {g.skills.join(", ")}
        </Text>
      ))}
    </View>
  );
}

/*
 * ExperienceSection
 * Linear stack; each entry = header row (title @ company | dates) + bullets.
 * @param items - work experience entries
 */
export function ExperienceSection({ items }: { items: ResumeExperience[] }) {
  if (!items.length) return null;
  return (
    <View>
      <SectionHeading heading={ResumeSectionHeading.EXPERIENCE} />
      {items.map((e, i) => (
        <View key={`${e.company}-${i}`} wrap={false}>
          <View style={s.entryHeaderRow}>
            <Text style={s.entryTitle}>
              {e.title} — {e.company}
            </Text>
            <Text style={s.entryMeta}>
              {e.startDate} – {e.isCurrent || !e.endDate ? "Present" : e.endDate}
            </Text>
          </View>
          {e.bullets.map((b, bi) => (
            <Text key={bi} style={s.bullet}>
              • {b}
            </Text>
          ))}
        </View>
      ))}
    </View>
  );
}

/*
 * EducationSection
 * Linear stack of institution + degree/major + date.
 * @param items - education entries
 */
export function EducationSection({ items }: { items: ResumeEducation[] }) {
  if (!items.length) return null;
  return (
    <View>
      <SectionHeading heading={ResumeSectionHeading.EDUCATION} />
      {items.map((ed, i) => (
        <View key={`${ed.institution}-${i}`}>
          <View style={s.entryHeaderRow}>
            <Text style={s.entryTitle}>{ed.institution}</Text>
            {ed.graduationDate ? (
              <Text style={s.entryMeta}>{ed.graduationDate}</Text>
            ) : null}
          </View>
          {ed.degree || ed.major ? (
            <Text style={s.paragraph}>
              {[ed.degree, ed.major].filter(Boolean).join(", ")}
            </Text>
          ) : null}
        </View>
      ))}
    </View>
  );
}

/*
 * CertificationsSection
 * Linear stack: name (issuer, date).
 * @param items - certifications
 */
export function CertificationsSection({
  items,
}: {
  items: ResumeCertification[];
}) {
  if (!items.length) return null;
  return (
    <View>
      <SectionHeading heading={ResumeSectionHeading.CERTIFICATIONS} />
      {items.map((c, i) => (
        <Text key={`${c.name}-${i}`} style={s.paragraph}>
          <Text style={s.entryTitle}>{c.name}</Text>
          {c.issuer ? ` — ${c.issuer}` : ""}
          {c.issueDate ? ` (${c.issueDate})` : ""}
        </Text>
      ))}
    </View>
  );
}

/*
 * ProjectsSection
 * Linear stack: title, optional link (selectable text), description.
 * @param items - projects
 */
export function ProjectsSection({ items }: { items: ResumeProject[] }) {
  if (!items.length) return null;
  return (
    <View>
      <SectionHeading heading={ResumeSectionHeading.PROJECTS} />
      {items.map((p, i) => (
        <View key={`${p.title}-${i}`}>
          <Text style={s.entryTitle}>{p.title}</Text>
          {p.url ? (
            <Link src={p.url} style={s.link}>
              {p.url}
            </Link>
          ) : null}
          {p.description ? (
            <Text style={s.paragraph}>{p.description}</Text>
          ) : null}
        </View>
      ))}
    </View>
  );
}
```

## Task 4: Assemble the `<Document>` template

**Files:**
- Create: `src/lib/resume/template/ResumeTemplate.tsx`

**ATS acceptance criteria for this task:** exactly one `<Page>`, single `<View>` column; sections rendered strictly in `RESUME_SECTION_ORDER`; header is name + headline + one contact `<Text>` line (contact row is inline meta, not a content column).

- [ ] **Step 1: Create the document component**

```tsx
import { Document, Page, Text, View } from "@react-pdf/renderer";
import { resumeStyles as s } from "@/lib/resume/template/styles";
import {
  SummarySection,
  SkillsSection,
  ExperienceSection,
  EducationSection,
  CertificationsSection,
  ProjectsSection,
} from "@/lib/resume/template/sections";
import type { ResumeDocument } from "@/modules/seeker/profile/types/resume-document.types";

/*
 * buildContactLine
 * Joins present contact fields into a single ATS-safe text line.
 * @param contact - ResumeDocument.contact
 * @returns "a@b.com · London · github.com/x" (present fields only)
 */
function buildContactLine(contact: ResumeDocument["contact"]): string {
  return [
    contact.email,
    contact.location,
    contact.websiteUrl,
    contact.linkedinUrl,
    contact.githubUrl,
  ]
    .filter(Boolean)
    .join("  ·  ");
}

/*
 * ResumeTemplate
 * The full react-pdf <Document>. Single page, single column, standard fonts,
 * real text only. Sections render in RESUME_SECTION_ORDER; each self-skips when
 * empty. This component is passed to renderToBuffer() by renderResumePdf().
 * @param doc - fully assembled ResumeDocument (Phase 4 output / Phase 5 input)
 */
export function ResumeTemplate({ doc }: { doc: ResumeDocument }) {
  const contactLine = buildContactLine(doc.contact);
  return (
    <Document title={`${doc.fullName} — Resume`} author={doc.fullName}>
      <Page size="A4" style={s.page}>
        <View>
          <Text style={s.name}>{doc.fullName}</Text>
          {doc.headline ? <Text style={s.headline}>{doc.headline}</Text> : null}
          {contactLine ? (
            <Text style={s.contactLine}>{contactLine}</Text>
          ) : null}
        </View>

        <SummarySection summary={doc.summary} />
        <SkillsSection groups={doc.skillGroups} />
        <ExperienceSection items={doc.experience} />
        <EducationSection items={doc.education} />
        <CertificationsSection items={doc.certifications} />
        <ProjectsSection items={doc.projects} />
      </Page>
    </Document>
  );
}
```

Note: section render order in JSX is authored to exactly match `RESUME_SECTION_ORDER`. Do not reorder.

```ts
/*
 * storeResumePdf
 * See interface doc above. Wraps the Buffer in a File (global in the Next.js
 * Node runtime) so it reuses lib/r2's uploadFile — no new S3 code paths.
 * @param seekerId - SeekerProfile id
 * @param pdf - rendered PDF buffer
 * @returns { key, url } — persist `key`, never the URL (URLs expire).
 */
export async function storeResumePdf(
  seekerId: string,
  pdf: Buffer,
): Promise<{ key: string; url: string }> {
  const key = R2_KEYS.resumeGenerated(seekerId);
  const file = new File([new Uint8Array(pdf)], "generated.pdf", {
    type: "application/pdf",
  });
  const { url } = await uploadFile(file, key);
  return { key, url };
}
```

```ts
import type {
  ResumeDocument,
  ResumeProject,
  ResumeSkillGroup,
} from "@/modules/seeker/profile/types/resume-document.types";

// (paste MapProfileInput interface here — see above)

/*
 * splitBullets
 * Splits a freeform description into ATS-friendly bullet lines: by newline, then
 * by "• " / "- " prefixes, trimming empties. Single-paragraph descriptions become
 * one bullet.
 * @param description - raw description text or null
 * @returns string[] bullet lines
 */
function splitBullets(description: string | null): string[] {
  if (!description?.trim()) return [];
  return description
    .split(/\r?\n|(?:^|\s)[•\-]\s+/g)
    .map((l) => l.trim())
    .filter(Boolean);
}

/*
 * groupSkillsByCategory
 * Collapses flat skills into ResumeSkillGroup[] preserving first-seen category
 * order. @param skills flat name+category list @returns grouped list
 */
function groupSkillsByCategory(
  skills: { name: string; category: string }[],
): ResumeSkillGroup[] {
  const order: string[] = [];
  const map = new Map<string, string[]>();
  for (const sk of skills) {
    if (!map.has(sk.category)) {
      map.set(sk.category, []);
      order.push(sk.category);
    }
    map.get(sk.category)!.push(sk.name);
  }
  return order.map((category) => ({ category, skills: map.get(category)! }));
}

/*
 * parseProjects
 * Coerces SeekerProfile.topProjects Json into typed ResumeProject[]. Drops
 * entries without a title. @param raw Json value @returns ResumeProject[]
 */
function parseProjects(raw: unknown): ResumeProject[] {
  if (!Array.isArray(raw)) return [];
  return raw
    .map((p): ResumeProject | null => {
      if (typeof p !== "object" || p === null) return null;
      const obj = p as { title?: unknown; url?: unknown; description?: unknown };
      if (typeof obj.title !== "string" || !obj.title.trim()) return null;
      return {
        title: obj.title,
        url: typeof obj.url === "string" ? obj.url : undefined,
        description:
          typeof obj.description === "string" ? obj.description : undefined,
      };
    })
    .filter((p): p is ResumeProject => p !== null);
}

/*
 * mapProfileToResumeDocument
 * See interface doc above.
 */
export function mapProfileToResumeDocument(
  input: MapProfileInput,
): ResumeDocument {
  const { profile } = input;
  return {
    fullName: input.fullName,
    headline: profile.headline ?? undefined,
    contact: {
      email: input.email ?? undefined,
      location: profile.location ?? undefined,
      websiteUrl: profile.websiteUrl ?? undefined,
      linkedinUrl: profile.linkedinUrl ?? undefined,
      githubUrl: profile.githubUrl ?? undefined,
    },
    summary: profile.bio ?? undefined,
    skillGroups: groupSkillsByCategory(input.skills),
    experience: input.experiences.map((e) => ({
      company: e.company,
      title: e.title,
      startDate: e.startDate,
      endDate: e.endDate,
      isCurrent: e.isCurrent,
      bullets: splitBullets(e.description),
    })),
    education: input.educations.map((ed) => ({
      institution: ed.institution,
      degree: ed.degree ?? undefined,
      major: ed.major ?? undefined,
      graduationDate: ed.graduationDate ?? undefined,
    })),
    certifications: input.certifications,
    projects: parseProjects(profile.topProjects),
  };
}
```
