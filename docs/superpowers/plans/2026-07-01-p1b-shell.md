## Goal

Stand up the seeker-only "apply faster" application shell: a nav-config-driven `AppSidebar` (generalized from the existing `RecruiterSidebar`, keeping the `SidebarContext`/collapse behavior), a new `src/app/(app)/layout.tsx` route-group shell guarded to `UserRole.SEEKER`, and stubbed routes for every sidebar destination (`/dashboard`, `/dashboard/browse`, `/dashboard/tracker`, `/dashboard/inbox`, `/dashboard/networking`, `/profile`, `/settings`, `/help`). A static bottom "credits + Upgrade" widget lands here (wired live in Phase 8). Finally, retarget the Phase-1a middleware redirect from `/seeker/dashboard` to the new `/dashboard`.

This is pure shell/navigation scaffolding. No data fetching, no AI, no billing logic — every route renders a placeholder screen. Phases 2–8 fill them in.

## Architecture

- **New route group `(app)`** at `src/app/(app)/` holds the pivot's seeker surface at top-level paths (`/dashboard`, `/profile`, `/settings`, `/help`). This is deliberately separate from the legacy `src/app/seeker/*` tree (untouched this phase) — the pivot moves to unprefixed paths per spec §8. Legacy `/seeker/*` stays intact behind its own layout until later phases retire it.
- **`AppSidebar`** is `RecruiterSidebar` generalized: same `SidebarContext` + collapse/toggle, same active-link styling, but driven by a `NavItem[]` prop instead of the hardcoded `RECRUITER_NAV_ITEMS`, and its CTA + bottom widget become configurable/passed-in rather than recruiter-specific.
- **`AppSidebarProvider`** wraps children with the context and renders `AppSidebar` beside `<main>` — mirrors `RecruiterSidebarProvider`.
- **`AppShellClient`** is the client boundary (mirrors `RecruiterLayoutClient`): mounts `ReactQueryProvider` + `AppSidebarProvider`. The route-group `layout.tsx` stays a server component that runs `AuthGuard(SEEKER)`.
- **Sidebar bottom widget** = a standalone `CreditsWidget` component rendering STATIC credits-left + Upgrade CTA now; Phase 8 swaps the static props for a live `useQuery`.

## Tech Stack

Next.js 14 App Router · TypeScript strict · TailwindCSS (brand tokens) · shadcn/ui `Button` · Lucide icons · TanStack Query provider (already exists at `@/providers/query-provider`). No new dependencies.

  /*
   * CreditsWidget
   * Bottom-of-sidebar widget: AI-credits-left progress + Upgrade CTA.

      <div className="px-4 pb-4">
        <div className="rounded-xl border border-brand-border bg-brand-surface p-3">
          <div className="mb-1 flex items-center justify-between">
            <span className="text-xs font-medium text-brand-foreground-muted">
              {planLabel} plan

  ```tsx
  interface AppSidebarProviderProps {
    navItems: NavItem[];
    homeHref: string;
    bottomSlot?: React.ReactNode;
    children: React.ReactNode;
  }

  /*
   * AppSidebarProvider
   * Provides sidebar collapse state and renders the config-driven AppSidebar
   * beside the page <main>. Generalized from RecruiterSidebarProvider.
   * @param navItems  - destinations to render
   * @param homeHref  - logo link target
   * @param bottomSlot- optional node pinned above the sidebar footer (e.g. CreditsWidget)
   */
  export function AppSidebarProvider({
    navItems,
    homeHref,
    bottomSlot,
    children,
  }: AppSidebarProviderProps) {
    const [collapsed, setCollapsed] = useState(false);
    const toggle = () => setCollapsed((c) => !c);

    return (
      <SidebarContext.Provider value={{ collapsed, toggle }}>
        <div className="flex min-h-screen">
          <AppSidebarInner
            collapsed={collapsed}
            navItems={navItems}
            homeHref={homeHref}
            bottomSlot={bottomSlot}
          />
          <main className="flex-1 min-w-0 overflow-x-hidden">{children}</main>
        </div>
      </SidebarContext.Provider>
    );
  }

  interface AppSidebarInnerProps {
    collapsed: boolean;
    navItems: NavItem[];
    homeHref: string;
    bottomSlot?: React.ReactNode;
  }

  const TOOLTIP_CLASSES =
    "rounded-lg border border-brand-border bg-white text-brand-foreground font-medium";

  /*
   * AppSidebarInner
   * Renders logo, config-driven navigation, and an optional bottom slot.
   * Disabled nav items render muted and non-navigating (placeholder surfaces).
   */
  function AppSidebarInner({
    collapsed,
    navItems,
    homeHref,
    bottomSlot,
  }: AppSidebarInnerProps) {
    const pathname = usePathname();

    return (
      <TooltipProvider delayDuration={0} skipDelayDuration={0}>
        <aside
          className={cn(
            "sticky top-0 h-screen bg-white border-r border-brand-border flex flex-col shrink-0 transition-[width] duration-200 ease-in-out overflow-hidden",
            collapsed ? "w-17" : "w-65",
          )}
        >
          {/* Logo + Toggle */}
          <div className={cn("px-5 py-5 flex items-center gap-3", collapsed && "justify-center px-0")}>
            {collapsed ? (
              <SidebarToggle />
            ) : (
              <>
                <Link href={homeHref} className="flex min-w-0 flex-1 items-center">
                  <Logo />
                </Link>
                <SidebarToggle className="shrink-0" />
              </>
            )}
          </div>

          <div className="mx-4 h-px bg-brand-border opacity-60" />

          {/* Navigation */}
          <nav className="flex-1 px-3 py-4 space-y-1 overflow-y-auto">
            {navItems.map((item) => (
              <AppSidebarLink
                key={item.href}
                item={item}
                collapsed={collapsed}
                isActive={
                  pathname === item.href || pathname.startsWith(`${item.href}/`)
                }
              />
            ))}
          </nav>

          {/* Bottom slot (e.g. CreditsWidget) */}
          {bottomSlot}
        </aside>
      </TooltipProvider>
    );
  }
  ```
- [ ] Add the single link renderer (DRY — one component handles collapsed + expanded + disabled, driven by data):
  ```tsx
  interface AppSidebarLinkProps {
    item: NavItem;
    collapsed: boolean;
    isActive: boolean;
  }

  /*
   * AppSidebarLink
   * One nav row. Disabled items render as a muted, non-navigating span with an
   * optional badge; enabled items render a Link with active-state styling.
   */
  function AppSidebarLink({ item, collapsed, isActive }: AppSidebarLinkProps) {
    const Icon = item.icon;

    const rowClasses = cn(
      "relative flex items-center py-2.5 text-sm transition-all border-l-[3px] rounded-r-md",
      collapsed ? "justify-center px-0 border-l-0 rounded-md mx-1" : "justify-between px-3",
      item.disabled
        ? "cursor-default border-transparent text-brand-foreground-secondary"
        : isActive
          ? "border-brand-primary bg-brand-primary/10 text-brand-primary font-semibold"
          : "border-transparent text-brand-foreground-muted hover:bg-brand-surface hover:text-brand-foreground font-medium",
    );

    const inner = (
      <>
        <div className={cn("flex items-center", collapsed ? "gap-0" : "gap-3")}>
          <Icon className="w-5 h-5 shrink-0" />
          {!collapsed && <span>{item.label}</span>}
        </div>
        {!collapsed && item.badge && (
          <span className="rounded bg-brand-surface px-1.5 py-0.5 text-[10px] font-semibold uppercase tracking-wide text-brand-foreground-secondary">
            {item.badge}
          </span>
        )}
      </>
    );

    const row = item.disabled ? (
      <span className={rowClasses} aria-disabled>
        {inner}
      </span>
    ) : (
      <Link href={item.href} className={rowClasses}>
        {inner}
      </Link>
    );

    if (!collapsed) return row;

    return (
      <Tooltip>
        <TooltipTrigger asChild>{row}</TooltipTrigger>
        <TooltipContent side="right" className={TOOLTIP_CLASSES}>
          {item.label}
          {item.badge ? ` · ${item.badge}` : ""}
        </TooltipContent>
      </Tooltip>
    );
  }
  ```
