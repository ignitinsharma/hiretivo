```typescript
export interface LockedEngineer {
  id: string;
  firstName: string;
  lastInitial: string;
  hiretivoScore: number;
  role: string;
  yearsExperience: number;
  skills: string[];
  salaryMinUsd: number;
  salaryMaxUsd: number;
  timezoneOverlapHours: number;
  availableFrom: string | null;
  isUnlocked: boolean;
  isSaved: boolean;
}

export interface UnlockedEngineer extends LockedEngineer {
  fullName: string;
  email: string;
  githubUrl: string | null;
  linkedinUrl: string | null;
  resumeUrl: string | null;
}

export type TalentEngineer = LockedEngineer | UnlockedEngineer;

export function isUnlocked(eng: TalentEngineer): eng is UnlockedEngineer {
  return eng.isUnlocked && "fullName" in eng;
}

export interface TalentFilters {
  skills?: string;        // comma-separated
  role?: string;
  salaryMax?: number;
  timezoneOverlapMin?: number;
  availableBefore?: string;
}

}
```

```typescript
"use client";

import { useInfiniteQuery, useMutation, useQueryClient } from "@tanstack/react-query";
import type { TalentEngineer, TalentFilters } from "../types/talent.types";

/*
 * useTalentPool
 * Infinite query for recruiter talent pool. Cursor-paginated.
 * staleTime: 30s (dynamic data — unlocks/saves change frequently).
 */
export function useTalentPool(filters: TalentFilters) {
  const params = new URLSearchParams();
  if (filters.skills) params.set("skills", filters.skills);
  if (filters.role) params.set("role", filters.role);
  if (filters.salaryMax) params.set("salaryMax", String(filters.salaryMax));
  if (filters.timezoneOverlapMin) params.set("timezoneOverlapMin", String(filters.timezoneOverlapMin));
  if (filters.availableBefore) params.set("availableBefore", filters.availableBefore);

  return useInfiniteQuery({
    queryKey: ["talent-pool", filters],
    queryFn: async ({ pageParam }: { pageParam: string | undefined }) => {
      if (pageParam) params.set("cursor", pageParam);
      const res = await fetch(`/api/recruiter/talent?${params.toString()}`);
      if (!res.ok) throw new Error(await res.text());
      const json = await res.json() as { data: { items: TalentEngineer[]; nextCursor: string | null } };
      return json.data;
    },
    initialPageParam: undefined as string | undefined,
    getNextPageParam: (lastPage) => lastPage.nextCursor ?? undefined,
    staleTime: 30_000,
  });
}

/*
 * useUnlockEngineer
 * Mutation to unlock an engineer profile. Invalidates talent-pool query on success.
 */
export function useUnlockEngineer() {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: async (engineerId: string) => {
      const res = await fetch(`/api/recruiter/talent/${engineerId}/unlock`, { method: "POST" });
      const json = await res.json();
      if (!res.ok) return { ok: false, code: json.code as string, resetAt: json.resetAt as string | null };
      return { ok: true, data: json.data };
    },
    onSuccess: (result) => {
      if (result.ok) qc.invalidateQueries({ queryKey: ["talent-pool"] });
    },
  });
}

/*
 * useSaveEngineer
 * Mutation to save or unsave an engineer. Invalidates talent-pool and saved-engineers queries.
 */
export function useSaveEngineer() {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: async ({ engineerId, save }: { engineerId: string; save: boolean }) => {
      const res = await fetch(`/api/recruiter/talent/${engineerId}/save`, {
        method: save ? "POST" : "DELETE",
      });
      if (!res.ok) throw new Error("Save failed");
      return res.json();
    },
    onSuccess: () => {
      qc.invalidateQueries({ queryKey: ["talent-pool"] });
      qc.invalidateQueries({ queryKey: ["saved-engineers"] });
    },
  });
}
```

```tsx
interface QuotaDisplayProps {
  used: number;
  total: number | null; // null = unlimited
  resetAt: Date | null;
}

/*
 * QuotaDisplay
 * Shows recruiter's unlock quota usage for the current billing period.
 * Renders "X / Y unlocks used" or "Unlimited unlocks" for SCALE plan.
 */
export function QuotaDisplay({ used, total, resetAt }: QuotaDisplayProps) {
  if (total === null) {
    return (
      <div className="flex items-center gap-1 text-sm text-brand-foreground-muted">
        <span className="text-brand-success font-medium">Unlimited</span>
        <span>unlocks</span>
      </div>
    );
  }

  const exhausted = used >= total;
  return (
    <div className="flex flex-col gap-0.5 text-sm">
      <div className="flex items-center gap-1">
        <span className={exhausted ? "font-medium text-brand-danger" : "font-medium text-brand-foreground"}>
          {used} / {total}
        </span>
        <span className="text-brand-foreground-muted">profile unlocks used this period</span>
      </div>
      {exhausted && resetAt && (
        <p className="text-xs text-brand-foreground-secondary">
          Resets {format(resetAt, "MMM d, yyyy")}
        </p>
      )}
    </div>
  );
}
```

```tsx
  isFreePlan: boolean;
  quotaExhausted: boolean;
  quotaResetAt?: string | null;

/*
 * EngineerCard
 * Renders one of 3 states based on unlock status and plan:
 * State 1 — Locked: blurred name, Unlock + Save buttons
 * State 2 — Unlocked: full PII, Email + Schedule buttons
 * State 3 — Quota exhausted: Unlock button grayed with tooltip
 * FREE plan: unlock attempt opens UpgradeModal instead of calling API
 */
export function EngineerCard({ engineer, isFreePlan, quotaExhausted, quotaResetAt }: EngineerCardProps) {
  const [upgradeModalOpen, setUpgradeModalOpen] = useState(false);

        <div className="flex items-start justify-between gap-2">
          <div>
            <div className="flex items-center gap-2">

                ✓ Verified

          </div>

        </div>

        <div className="flex flex-wrap gap-x-4 gap-y-1 text-sm text-brand-foreground-muted">
          <span>{engineer.yearsExperience} yrs exp</span>
          <span>${engineer.salaryMinUsd.toLocaleString()}–${engineer.salaryMaxUsd.toLocaleString()} /yr</span>
          <span>{engineer.timezoneOverlapHours}h overlap</span>

        </div>

        {/* Skills */}
        {engineer.skills.length > 0 && (
          <div className="flex flex-wrap gap-1.5">
            {engineer.skills.slice(0, 6).map((skill) => (
```

```tsx
import { auth } from "@/auth";
import { redirect } from "next/navigation";
import { UserRole } from "@/constants/enums";
import { prisma } from "@/lib/prisma";
import { canAccessTalentPool, getUnlockQuota } from "@/lib/billing/plan-access";
import { generatePageMetadata } from "@/lib/metadata";
import { TalentPoolClient } from "@/modules/recruiter/talent/components/TalentPoolClient";

export const metadata = generatePageMetadata({
  title: "Talent Pool",
  description: "Browse and unlock verified Indian engineers.",
});

/*
 * TalentPoolPage
 * Server component: loads recruiter plan, quota used, subscription data.
 * Passes to TalentPoolClient for interactive browse UX.
 */
export default async function TalentPoolPage() {
  const session = await auth();
  if (!session?.user?.id) redirect("/auth/signin");
  if (session.user.role !== UserRole.RECRUITER) redirect("/seeker/dashboard");

  const recruiter = await prisma.user.findUnique({
    where: { id: session.user.id },
    select: {
      plan: true,
      subscription: { select: { currentPeriodStart: true, currentPeriodEnd: true } },
    },
  });

  const plan = recruiter?.plan ?? "FREE";
  const isFreePlan = !canAccessTalentPool(plan);
  const totalQuota = getUnlockQuota(plan);

  let usedQuota = 0;
  if (!isFreePlan && totalQuota !== null) {
    const periodStart = recruiter?.subscription?.currentPeriodStart ?? new Date(new Date().getFullYear(), new Date().getMonth(), 1);
    usedQuota = await prisma.engineerUnlock.count({
      where: {
        recruiterId: session.user.id,
        source: "talent_pool",
        unlockedAt: { gte: periodStart },
      },
    });
  }

  return (
    <TalentPoolClient
      isFreePlan={isFreePlan}
      totalQuota={totalQuota}
      usedQuota={usedQuota}
      quotaResetAt={recruiter?.subscription?.currentPeriodEnd?.toISOString() ?? null}
    />
  );
}
```

```tsx
"use client";

import { useState } from "react";
import { EngineerCard } from "./EngineerCard";
import { TalentFilters } from "./TalentFilters";
import { QuotaDisplay } from "./QuotaDisplay";
import { Button } from "@/components/ui/button";
import { useTalentPool } from "../queries/use-talent-pool";
import type { TalentFilters as TalentFiltersType } from "../types/talent.types";

interface TalentPoolClientProps {
  isFreePlan: boolean;
  totalQuota: number | null;
  usedQuota: number;
  quotaResetAt: string | null;
}

/*
 * TalentPoolClient
 * Client component for talent pool browse page.
 * Handles filter state, infinite scroll pagination, and quota display.
 */
export function TalentPoolClient({ isFreePlan, totalQuota, usedQuota, quotaResetAt }: TalentPoolClientProps) {
  const [filters, setFilters] = useState<TalentFiltersType>({});
  const { data, fetchNextPage, hasNextPage, isFetchingNextPage, isLoading, isError } = useTalentPool(filters);

  const engineers = data?.pages.flatMap((p) => p.items) ?? [];
  const quotaExhausted = totalQuota !== null && usedQuota >= totalQuota;

  return (
    <div className="flex flex-col gap-6">
      <div className="flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-between">
        <div>
          <h1 className="font-bold text-3xl text-brand-foreground">Talent Pool</h1>
          <p className="text-brand-foreground-muted mt-1">Browse verified Indian engineers</p>
        </div>
        {!isFreePlan && (
          <QuotaDisplay
            used={usedQuota}
            total={totalQuota}
            resetAt={quotaResetAt ? new Date(quotaResetAt) : null}
          />
        )}
      </div>

      <TalentFilters filters={filters} onChange={setFilters} />

      {isLoading && (
        <div className="grid grid-cols-1 gap-4 md:grid-cols-2 xl:grid-cols-3">
          {Array.from({ length: 6 }).map((_, i) => (
            <div key={i} className="h-56 rounded-xl border border-brand-border bg-brand-surface animate-pulse" />
          ))}
        </div>
      )}

      {isError && (
        <div className="rounded-xl border border-brand-border bg-white p-6 text-center">
          <p className="text-brand-danger font-medium">Failed to load talent pool</p>
          <p className="text-sm text-brand-foreground-muted mt-1">Please refresh the page</p>
        </div>
      )}

      {!isLoading && !isError && engineers.length === 0 && (
        <div className="rounded-xl border border-brand-border bg-white p-10 text-center">
          <p className="font-semibold text-brand-foreground">No engineers match your filters</p>
          <p className="text-sm text-brand-foreground-muted mt-1">Try adjusting your filters</p>
        </div>
      )}

      {engineers.length > 0 && (
        <div className="grid grid-cols-1 gap-4 md:grid-cols-2 xl:grid-cols-3">
          {engineers.map((eng) => (
            <EngineerCard
              key={eng.id}
              engineer={eng}
              isFreePlan={isFreePlan}
              quotaExhausted={quotaExhausted}
              quotaResetAt={quotaResetAt}
            />
          ))}
        </div>
      )}

      {hasNextPage && (
        <div className="flex justify-center">
          <Button variant="secondary" onClick={() => fetchNextPage()} disabled={isFetchingNextPage}>
            {isFetchingNextPage ? "Loading…" : "Load more"}
          </Button>
        </div>
      )}
    </div>
  );
}
```
