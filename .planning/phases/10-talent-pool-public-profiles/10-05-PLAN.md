```typescript
"use client";

import { useQuery, useMutation, useQueryClient } from "@tanstack/react-query";

interface WhoViewedMeEntry {
  companyName: string;
  companyDomain: string;
  unlockedAt: string;
  source: "talent_pool" | "application_auto";
}

interface BlockedDomainEntry {
  domain: string;
  blockedAt: string;
}

interface ProfileVisibilityData {
  blockedDomains: BlockedDomainEntry[];
  whoViewedMe: WhoViewedMeEntry[];
}

/*
 * useProfileVisibility
 * Fetches blocked recruiter domains and who-viewed-me from
 * GET /api/engineer/block-recruiter.
 * staleTime: 30s.
 */
export function useProfileVisibility() {
  return useQuery({
    queryKey: ["profile-visibility"],
    queryFn: async (): Promise<ProfileVisibilityData> => {
      const res = await fetch("/api/engineer/block-recruiter");
      if (!res.ok) throw new Error(await res.text());
      const json = await res.json() as { data: ProfileVisibilityData };
      return json.data;
    },
    staleTime: 30_000,
  });
}

/*
 * useHideProfile
 * Mutation to set or clear hiddenUntil via POST /api/engineer/hide-profile.
 * Invalidates profile-visibility on success.
 */
export function useHideProfile() {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: async (params: { hide: boolean; until?: string }) => {
      const res = await fetch("/api/engineer/hide-profile", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(params),
      });
      if (!res.ok) throw new Error(await res.text());
      return res.json() as Promise<{ data: { hidden: boolean; hiddenUntil: string | null } }>;
    },
    onSuccess: () => {
      qc.invalidateQueries({ queryKey: ["profile-visibility"] });
      qc.invalidateQueries({ queryKey: ["seeker-settings"] });
    },
  });
}

/*
 * useBlockRecruiter
 * Mutation to block a recruiter domain via POST /api/engineer/block-recruiter.
 * Invalidates profile-visibility on success.
 */
export function useBlockRecruiter() {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: async (recruiterDomain: string) => {
      const res = await fetch("/api/engineer/block-recruiter", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ recruiterDomain }),
      });
      if (!res.ok) {
        const json = await res.json() as { error: string };
        throw new Error(json.error ?? "Failed to block");
      }
      return res.json() as Promise<{ data: { blocked: string } }>;
    },
    onSuccess: () => {
      qc.invalidateQueries({ queryKey: ["profile-visibility"] });
    },
  });
}
```

```tsx
"use client";

import { Button } from "@/components/ui/button";
import { format, addDays } from "date-fns";
import { useHideProfile } from "../queries/use-profile-visibility";
import { EyeOff, Eye } from "lucide-react";

interface HideProfileToggleProps {
  hiddenUntil: string | null; // ISO string from user record
}

/*
 * HideProfileToggle
 * Toggle for "Hide my profile temporarily" in engineer settings.
 * hide=true: sets hiddenUntil = now + 30 days. hide=false: clears hiddenUntil.
 * When hidden: shows "Profile hidden until [date]" with "Make visible again" CTA.
 */
export function HideProfileToggle({ hiddenUntil }: HideProfileToggleProps) {
  const { mutate: hideProfile, isPending } = useHideProfile();

  const isHidden = hiddenUntil !== null && new Date(hiddenUntil) > new Date();
  const hiddenUntilDate = hiddenUntil ? new Date(hiddenUntil) : null;

  return (
    <div className="rounded-xl border border-brand-border bg-white p-5 flex flex-col gap-3">
      <div className="flex items-start justify-between gap-4">
        <div>
          <h3 className="font-semibold text-brand-foreground flex items-center gap-2">
            <EyeOff className="h-4 w-4 text-brand-foreground-muted" />
            Hide my profile temporarily
          </h3>
          <p className="text-sm text-brand-foreground-muted mt-0.5">
            Your profile will be removed from the talent pool and your public URL will return 404.
            Active applications and interviews continue normally.
          </p>
        </div>
      </div>

      {isHidden && hiddenUntilDate ? (
        <div className="flex flex-col gap-2 rounded-lg border border-brand-border bg-brand-surface p-3">
          <p className="text-sm font-medium text-brand-foreground flex items-center gap-1.5">
            <EyeOff className="h-3.5 w-3.5" />
            Hidden until {format(hiddenUntilDate, "MMM d, yyyy")}
          </p>
          <Button
            variant="secondary"
            size="sm"
            className="self-start"
            onClick={() => hideProfile({ hide: false })}
            disabled={isPending}
          >
            <Eye className="h-3.5 w-3.5 mr-1.5" />
            {isPending ? "Updating…" : "Make profile visible again"}
          </Button>
        </div>
      ) : (
        <Button
          variant="ghost"
          size="sm"
          className="self-start border border-brand-border"
          onClick={() => hideProfile({ hide: true })}
          disabled={isPending}
        >
          <EyeOff className="h-3.5 w-3.5 mr-1.5" />
          {isPending ? "Hiding…" : `Hide for 30 days (until ${format(addDays(new Date(), 30), "MMM d")})`}
        </Button>
      )}
    </div>
  );
}
```

```tsx
"use client";

import { useState } from "react";
import { Button } from "@/components/ui/button";
import { format } from "date-fns";
import { useProfileVisibility, useBlockRecruiter } from "../queries/use-profile-visibility";
import { Shield, Eye } from "lucide-react";

/*
 * WhoViewedMeSection
 * Lists recruiters who unlocked the engineer's profile.
 * Each entry shows company name, domain, unlock date, source, and a "Block this company" button.
 */
export function WhoViewedMeSection() {
  const { data, isLoading, isError } = useProfileVisibility();
  const { mutate: blockRecruiter, isPending: blocking } = useBlockRecruiter();
  const [blockedNow, setBlockedNow] = useState<Set<string>>(new Set());

  if (isLoading) {
    return (
      <div className="rounded-xl border border-brand-border bg-white p-5">
        <div className="h-4 w-32 rounded bg-brand-surface animate-pulse mb-3" />
        {[1, 2].map((i) => <div key={i} className="h-12 rounded-lg bg-brand-surface animate-pulse mb-2" />)}
      </div>
    );
  }

  if (isError) {
    return (
      <div className="rounded-xl border border-brand-border bg-white p-5">
        <p className="text-sm text-brand-danger">Failed to load viewer history</p>
      </div>
    );
  }

  const viewers = data?.whoViewedMe ?? [];

  return (
    <div className="rounded-xl border border-brand-border bg-white p-5 flex flex-col gap-4">
      <div>
        <h3 className="font-semibold text-brand-foreground flex items-center gap-2">
          <Eye className="h-4 w-4 text-brand-foreground-muted" />
          Who viewed me
        </h3>
        <p className="text-sm text-brand-foreground-muted mt-0.5">
          Recruiters who unlocked your profile. You can block companies from finding you.
        </p>
      </div>

      {viewers.length === 0 ? (
        <p className="text-sm text-brand-foreground-secondary">No one has unlocked your profile yet.</p>
      ) : (
        <div className="flex flex-col gap-2">
          {viewers.map((viewer, i) => {
            const isBlockedNow = blockedNow.has(viewer.companyDomain);
            return (
              <div key={i} className="flex items-center justify-between gap-3 rounded-lg border border-brand-border bg-brand-surface px-4 py-3">
                <div className="min-w-0">
                  <p className="font-medium text-sm text-brand-foreground truncate">{viewer.companyName}</p>
                  <p className="text-xs text-brand-foreground-secondary">
                    {viewer.source === "talent_pool" ? "Viewed in Talent Pool" : "Viewed application"} · {format(new Date(viewer.unlockedAt), "MMM d, yyyy")}
                  </p>
                </div>
                {isBlockedNow ? (
                  <span className="text-xs text-brand-foreground-secondary flex items-center gap-1">
                    <Shield className="h-3.5 w-3.5" /> Blocked
                  </span>
                ) : (
                  <Button
                    variant="ghost"
                    size="sm"
                    className="shrink-0 text-brand-danger hover:bg-brand-danger/10"
                    disabled={blocking}
                    onClick={() => {
                      blockRecruiter(viewer.companyDomain, {
                        onSuccess: () => setBlockedNow((prev) => new Set(prev).add(viewer.companyDomain)),
                      });
                    }}
                  >
                    Block
                  </Button>
                )}
              </div>
            );
          })}
        </div>
      )}
    </div>
  );
}
```
