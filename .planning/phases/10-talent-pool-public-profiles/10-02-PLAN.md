```typescript
import { auth } from "@/auth";
import { UserRole } from "@/constants/enums";
import { prisma } from "@/lib/prisma";
import { writeRateLimiter } from "@/lib/rate-limit";
import { NextRequest, NextResponse } from "next/server";

/*
 * POST /api/recruiter/talent/[id]/save
 * Saves an engineer to recruiter's saved list. Idempotent — no quota deducted.
 * Returns { saved: true } on success or if already saved.
 */
export async function POST(req: NextRequest, { params }: { params: { id: string } }) {
  const session = await auth();
  if (!session?.user?.id) return new Response("Unauthorized", { status: 401 });
  if (session.user.role !== UserRole.RECRUITER) return new Response("Forbidden", { status: 403 });

  const { success } = await writeRateLimiter.limit(session.user.id);
  if (!success) return new Response("Too Many Requests", { status: 429 });

  await prisma.savedEngineer.upsert({
    where: { recruiterId_engineerId: { recruiterId: session.user.id, engineerId: params.id } },
    update: {},
    create: { recruiterId: session.user.id, engineerId: params.id },
  });

  return NextResponse.json({ data: { saved: true } });
}

/*
 * DELETE /api/recruiter/talent/[id]/save
 * Removes an engineer from recruiter's saved list.
 * Returns { saved: false } on success or if not saved.
 */
export async function DELETE(req: NextRequest, { params }: { params: { id: string } }) {
  const session = await auth();
  if (!session?.user?.id) return new Response("Unauthorized", { status: 401 });
  if (session.user.role !== UserRole.RECRUITER) return new Response("Forbidden", { status: 403 });

  const { success } = await writeRateLimiter.limit(session.user.id);
  if (!success) return new Response("Too Many Requests", { status: 429 });

  await prisma.savedEngineer.deleteMany({
    where: { recruiterId: session.user.id, engineerId: params.id },
  });

  return NextResponse.json({ data: { saved: false } });
}
```

```typescript
import { auth } from "@/auth";
import { UserRole } from "@/constants/enums";
import { prisma } from "@/lib/prisma";
import { writeRateLimiter } from "@/lib/rate-limit";
import { sendEmail } from "@/lib/resend";
import { z } from "zod";
import { NextRequest, NextResponse } from "next/server";

const InviteSchema = z.object({
  jobId: z.string().min(1),
  note: z.string().max(500).optional(),
});

/*
 * POST /api/recruiter/talent/[id]/invite
 * Creates a RECRUITER_INITIATED application (interview invite from talent pool).
 * Requires EngineerUnlock to exist for this recruiter+engineer pair.
 * Returns 400 NO_OPEN_JOBS if recruiter has no OPEN jobs.
 * Returns 409 ALREADY_APPLIED if Application already exists for this job+engineer.
 */
export async function POST(req: NextRequest, { params }: { params: { id: string } }) {
  const session = await auth();
  if (!session?.user?.id) return new Response("Unauthorized", { status: 401 });
  if (session.user.role !== UserRole.RECRUITER) return new Response("Forbidden", { status: 403 });

  const { success } = await writeRateLimiter.limit(session.user.id);
  if (!success) return new Response("Too Many Requests", { status: 429 });

  const body = await req.json().catch(() => ({}));
  const parsed = InviteSchema.safeParse(body);
  if (!parsed.success) {
    return NextResponse.json({ error: parsed.error.flatten() }, { status: 400 });
  }
  const { jobId } = parsed.data;

  // Must have existing unlock
  const unlock = await prisma.engineerUnlock.findUnique({
    where: { recruiterId_engineerId: { recruiterId: session.user.id, engineerId: params.id } },
  });
  if (!unlock) {
    return NextResponse.json({ code: "NOT_UNLOCKED" }, { status: 403 });
  }

  // Check job is OPEN and owned by recruiter
  const job = await prisma.job.findFirst({
    where: { id: jobId, recruiterId: session.user.id, status: "OPEN" },
    select: { id: true, title: true },
  });
  if (!job) {
    // Check if recruiter has ANY open jobs
    const openJobCount = await prisma.job.count({ where: { recruiterId: session.user.id, status: "OPEN" } });
    if (openJobCount === 0) {
      return NextResponse.json({ code: "NO_OPEN_JOBS" }, { status: 400 });
    }
    return NextResponse.json({ error: "Job not found or not open" }, { status: 404 });
  }

  // Check duplicate application
  const existing = await prisma.application.findUnique({
    where: { jobId_seekerId: { jobId, seekerId: params.id } },
    select: { id: true },
  });
  if (existing) {
    return NextResponse.json({ code: "ALREADY_APPLIED", applicationId: existing.id }, { status: 409 });
  }

  // Load engineer for email
  const engineer = await prisma.user.findUnique({
    where: { id: params.id },
    select: { name: true, email: true },
  });
  const recruiter = await prisma.user.findUnique({
    where: { id: session.user.id },
    select: { recruiterProfile: { select: { companyName: true } } },
  });

  // Create application
  const application = await prisma.application.create({
    data: {
      jobId,
      seekerId: params.id,
      source: "RECRUITER_INITIATED",
      status: "APPLIED",
      matchScore: 0, // Phase 11 will compute real score
    },
  });

  // Fire-and-forget invite email
  const companyName = recruiter?.recruiterProfile?.companyName ?? "A recruiter";
  sendEmail({
    from: "Hiretivo <noreply@hiretivo.com>",
    to: engineer?.email ?? "",
    subject: `You've been invited to interview at ${companyName}`,
    html: `<p>Hi ${engineer?.name ?? "Engineer"},</p><p>You've been invited to interview at <strong>${companyName}</strong> for the role of <strong>${job.title}</strong>.</p><p><a href="https://hiretivo.com/seeker/applications">View invitation →</a></p>`,
    text: `Hi ${engineer?.name ?? "Engineer"},\n\nYou've been invited to interview at ${companyName} for ${job.title}.\n\nhttps://hiretivo.com/seeker/applications`,
  }).catch(() => { /* non-critical */ });

  return NextResponse.json({ data: { applicationId: application.id } }, { status: 201 });
}
```

```typescript
import { auth } from "@/auth";
import { UserRole } from "@/constants/enums";
import { prisma } from "@/lib/prisma";
import { writeRateLimiter } from "@/lib/rate-limit";
import { addDays } from "date-fns";
import { z } from "zod";
import { NextRequest, NextResponse } from "next/server";

const HideProfileSchema = z.object({
  hide: z.boolean(),
  until: z.string().datetime().optional(), // ISO datetime; if omitted, defaults to +30 days
});

/*
 * POST /api/engineer/hide-profile
 * Sets or clears hiddenUntil on the engineer's User record.
 * hide=true: sets hiddenUntil = until ?? addDays(now, 30).
 * hide=false: clears hiddenUntil = null.
 * Effect is immediate — profile disappears from talent pool and /e/[slug] returns 404.
 */
export async function POST(req: NextRequest) {
  const session = await auth();
  if (!session?.user?.id) return new Response("Unauthorized", { status: 401 });
  if (session.user.role !== UserRole.SEEKER) return new Response("Forbidden", { status: 403 });

  const { success } = await writeRateLimiter.limit(session.user.id);
  if (!success) return new Response("Too Many Requests", { status: 429 });

  const body = await req.json().catch(() => ({}));
  const parsed = HideProfileSchema.safeParse(body);
  if (!parsed.success) {
    return NextResponse.json({ error: parsed.error.flatten() }, { status: 400 });
  }

  const hiddenUntil = parsed.data.hide
    ? (parsed.data.until ? new Date(parsed.data.until) : addDays(new Date(), 30))
    : null;

  await prisma.user.update({
    where: { id: session.user.id },
    data: { hiddenUntil },
  });

  return NextResponse.json({ data: { hidden: parsed.data.hide, hiddenUntil: hiddenUntil?.toISOString() ?? null } });
}
```

```typescript
import { auth } from "@/auth";
import { UserRole } from "@/constants/enums";
import { prisma } from "@/lib/prisma";
import { writeRateLimiter, dataFetchRateLimiter, getRateLimitIp } from "@/lib/rate-limit";
import { z } from "zod";
import { NextRequest, NextResponse } from "next/server";

// Generic email domains that cannot be blocked as company domains
const GENERIC_DOMAINS = new Set([
  "gmail.com", "outlook.com", "yahoo.com", "hotmail.com", "icloud.com",
  "live.com", "msn.com", "aol.com", "protonmail.com", "mail.com",
]);

/*
 * normalizeDomain
 * Strips subdomains, lowercases, and validates the result.
 * Returns normalized domain string or null if invalid/generic.
 * Example: "us.acme.com" -> "acme.com"
 */
function normalizeDomain(raw: string): string | null {
  try {
    const parts = raw.toLowerCase().trim().split(".");
    if (parts.length < 2) return null;
    // Take last two parts: "acme.com" from "us.acme.com"
    const normalized = `${parts[parts.length - 2]}.${parts[parts.length - 1]}`;
    if (GENERIC_DOMAINS.has(normalized)) return null;
    return normalized;
  } catch {
    return null;
  }
}

const BlockSchema = z.object({
  recruiterDomain: z.string().min(3).max(100),
});

/*
 * POST /api/engineer/block-recruiter
 * Creates BlockedRecruiter(engineerId, recruiterDomain).
 * Normalizes domain before storing. Rejects generic email domains.
 * Idempotent — existing block record is a no-op.
 */
export async function POST(req: NextRequest) {
  const session = await auth();
  if (!session?.user?.id) return new Response("Unauthorized", { status: 401 });
  if (session.user.role !== UserRole.SEEKER) return new Response("Forbidden", { status: 403 });

  const { success } = await writeRateLimiter.limit(session.user.id);
  if (!success) return new Response("Too Many Requests", { status: 429 });

  const body = await req.json().catch(() => ({}));
  const parsed = BlockSchema.safeParse(body);
  if (!parsed.success) {
    return NextResponse.json({ error: parsed.error.flatten() }, { status: 400 });
  }

  const domain = normalizeDomain(parsed.data.recruiterDomain);
  if (!domain) {
    return NextResponse.json({ error: "Invalid or generic domain" }, { status: 400 });
  }

  await prisma.blockedRecruiter.upsert({
    where: { engineerId_recruiterDomain: { engineerId: session.user.id, recruiterDomain: domain } },
    update: {},
    create: { engineerId: session.user.id, recruiterDomain: domain },
  });

  return NextResponse.json({ data: { blocked: domain } });
}

/*
 * GET /api/engineer/block-recruiter
 * Returns:
 * - blockedDomains: list of currently blocked recruiter domains
 * - whoViewedMe: list of recruiters who unlocked this engineer (from EngineerUnlock)
 *   Each entry: { companyName, companyDomain, unlockedAt }
 */
export async function GET(req: NextRequest) {
  const session = await auth();
  if (!session?.user?.id) return new Response("Unauthorized", { status: 401 });
  if (session.user.role !== UserRole.SEEKER) return new Response("Forbidden", { status: 403 });

  const ip = getRateLimitIp(req);
  const { success } = await dataFetchRateLimiter.limit(ip);
  if (!success) return new Response("Too Many Requests", { status: 429 });

  const [blocked, unlocks] = await Promise.all([
    prisma.blockedRecruiter.findMany({
      where: { engineerId: session.user.id },
      orderBy: { createdAt: "desc" },
      select: { recruiterDomain: true, createdAt: true },
    }),
    prisma.engineerUnlock.findMany({
      where: { engineerId: session.user.id },
      orderBy: { unlockedAt: "desc" },
      take: 50,
      select: {
        unlockedAt: true, source: true,
        recruiter: {
          select: {
            recruiterProfile: { select: { companyName: true, website: true } },
          },
        },
      },
    }),
  ]);

  const whoViewedMe = unlocks.map((u) => {
    const website = u.recruiter.recruiterProfile?.website ?? "";
    const domain = normalizeDomain(website.replace(/^https?:\/\//, "").split("/")[0] ?? "") ?? website;
    return {
      companyName: u.recruiter.recruiterProfile?.companyName ?? "Unknown",
      companyDomain: domain,
      unlockedAt: u.unlockedAt.toISOString(),
      source: u.source,
    };
  });

  return NextResponse.json({
    data: {
      blockedDomains: blocked.map((b) => ({ domain: b.recruiterDomain, blockedAt: b.createdAt.toISOString() })),
      whoViewedMe,
    },
  });
}
```
