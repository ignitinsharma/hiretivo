async function handleBoostPaymentSucceeded(
  data: PaymentSucceededWebhookEvent['data'],
): Promise<void> {
  const meta = data.metadata as Record<string, string> | undefined;
  if (!meta?.userId || !meta?.jobId || meta?.type !== 'boost') return;

  const { userId, jobId } = meta;
  const startsAt = new Date();
  const expiresAt = new Date(startsAt.getTime() + BOOST_DURATION_MS);

  await prisma.boostedJob.upsert({
    where: { jobId },
    create: {
      jobId,
      recruiterId: userId,
      paymentId: data.payment_id,
      startsAt,
      expiresAt,
    },
    update: {
      recruiterId: userId,
      paymentId: data.payment_id,
      startsAt,
      expiresAt,
    },
  });
}
```

export function BoostJobButton({ jobId, isAlreadyBoosted, boostExpiresAt }: BoostJobButtonProps) {
  const [isLoading, setIsLoading] = useState(false);

  const handleBoost = async () => {
    setIsLoading(true);
    try {
      const res = await fetch('/api/billing/boost-checkout', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ jobId }),
      });
      const json = await res.json() as { data?: { url: string }; error?: string };
      if (!res.ok || !json.data?.url) {
        throw new Error(json.error ?? 'Failed to start boost');
      }
      window.location.href = json.data.url;
    } catch (err) {
      toast.error(err instanceof Error ? err.message : 'Failed to start boost. Try again.');
      setIsLoading(false);
    }
  };

  if (isAlreadyBoosted && boostExpiresAt) {
    return (
      <div className="rounded-xl border border-green-200 bg-green-50 p-5 space-y-3">
        <div className="flex items-center gap-2">
          <Zap className="h-4 w-4 text-green-600" aria-hidden="true" />
          <p className="text-sm font-semibold text-green-800">
            Boosted — active until {formatBoostExpiry(boostExpiresAt)}
          </p>
        </div>
        <Button disabled className="w-full" variant="outline">Extend Boost</Button>
      </div>
    );
  }

  return (
    <div className="rounded-xl border border-slate-200 bg-slate-50 p-5 space-y-4">
      <div>
        <h3 className="text-sm font-bold text-slate-900 flex items-center gap-2">
          <Zap className="h-4 w-4 text-indigo-500" aria-hidden="true" />
          Boost this Job
        </h3>
        <p className="text-xs text-slate-500 mt-1">Pin at the top of matched seeker feeds for 7 days — $9 one-time.</p>
      </div>
      <ul className="space-y-1">
        {BOOST_FEATURES.map((feature) => (
          <li key={feature} className="text-xs text-slate-600 flex items-center gap-1.5">
            <span className="text-green-500" aria-hidden="true">✓</span> {feature}
          </li>
        ))}
      </ul>
      <Button
        onClick={handleBoost}
        disabled={isLoading}
        className="w-full bg-indigo-600 hover:bg-indigo-700 text-white font-semibold"
        aria-label="Boost this job listing"
      >
        {isLoading ? 'Redirecting...' : 'Boost for $9'}
      </Button>
    </div>
  );
}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Client → POST /api/billing/boost-checkout | Untrusted recruiter request body with jobId |
| DodoPayments → POST /api/billing/webhook | External webhook from DodoPayments servers; signature verified by existing dodo.webhooks.unwrap() |
| Webhook metadata → prisma.boostedJob.upsert | Metadata values (userId, jobId) from webhook event data — must not be trusted without ownership validation |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-13-13 | Tampering | POST /api/billing/boost-checkout — recruiter boosts another recruiter's job | mitigate | `prisma.job.findFirst({ where: { id: jobId, recruiterId: session.user.id } })` returns 404 if not owned; prevents cross-recruiter boost injection (ASVS V4.2.1) |
| T-13-14 | Spoofing | Webhook replay — boost activated multiple times | mitigate | `prisma.boostedJob.upsert` (not create) is idempotent; replaying the same payment.succeeded event just resets expiresAt to the same value |
| T-13-15 | Tampering | Webhook metadata.userId/jobId crafted to boost arbitrary job | mitigate | Webhook signature is verified by existing `dodo.webhooks.unwrap()` at handler entry; forged webhooks without valid DodoPayments HMAC are rejected with 400 |
| T-13-16 | Information Disclosure | Boost checkout route returns job details to non-owners | mitigate | Route returns 404 for unowned jobs (not 403) — prevents leaking job existence to other recruiters |
| T-13-17 | Elevation of Privilege | Expired BoostedJob still pinning in seeker feed | mitigate | Feed query filters `expiresAt: { gt: now }` strictly; only active boosts are prepended (ASVS V4.1.3) |
| T-13-18 | Denial of Service | Boost checkout spammed to generate excessive DodoPayments API calls | mitigate | Rate limit: 5 boost attempts/min per recruiter via Upstash; DodoPayments also has their own rate limits |
