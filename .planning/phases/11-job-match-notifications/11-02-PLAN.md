/*
 * getJobPostingLimit
 * Returns the max number of OPEN jobs allowed for a plan tier.
 * Returns null for unlimited (GROWTH, SCALE, ENTERPRISE).
 * FREE = 0 (cannot post), STARTER = 2, PRO = 10 (legacy), others null.
 */
export function getJobPostingLimit(plan: string | null | undefined): number | null {
  switch (plan) {
    case PlanTier.FREE:    return 0
    case PlanTier.STARTER: return 2
    case PlanTier.GROWTH:  return null   // unlimited
    case PlanTier.SCALE:   return null   // unlimited
    case PlanTier.PRO:     return 10     // legacy
    case PlanTier.ENTERPRISE: return null // legacy
    default:               return 0      // unknown = treat as FREE
  }
}

/*
 * canPostJob
 * Returns true if the recruiter can publish a new OPEN job.
 * @param plan - current plan tier
 * @param currentActiveCount - current count of OPEN jobs for this recruiter
 */
export function canPostJob(plan: string | null | undefined, currentActiveCount: number): boolean {
  const limit = getJobPostingLimit(plan)
  if (limit === null) return true          // unlimited
  return currentActiveCount < limit
}
```

/*
 * handleJobDowngrade
 * When a recruiter's plan is downgraded, auto-drafts excess OPEN jobs
 * beyond the new plan's job posting limit. Most-recently-created jobs
 * are kept OPEN; oldest excess jobs become DRAFT.
 */
async function handleJobDowngrade(userId: string, newPlan: PlanTier): Promise<void> {
  const { getJobPostingLimit } = await import('@/lib/billing/plan-access')
  const newQuota = getJobPostingLimit(newPlan)
  if (newQuota === null) return // unlimited — nothing to draft

  const openJobs = await prisma.job.findMany({
    where: { recruiterId: userId, status: JobStatus.OPEN },
    orderBy: { createdAt: 'desc' },
    select: { id: true },
  })

  const excess = openJobs.slice(newQuota)
  if (excess.length === 0) return

  await prisma.job.updateMany({
    where: { id: { in: excess.map(j => j.id) } },
    data: { status: JobStatus.DRAFT },
  })
  console.info(`[billing] Downgrade: auto-drafted ${excess.length} excess jobs for user ${userId}`)
}
```
