**Step 2 — Create email template `src/lib/email/job-match-notification.ts`:**

```typescript
import { resend } from '@/lib/resend'

export interface JobMatchItem {
  id: string
  title: string
  companyName: string
  matchScore: number
}

/*
 * sendJobMatchNotification
 * Sends a daily digest email to a VERIFIED engineer with new matching jobs.
 * @param email - engineer's email address
 * @param name - engineer's display name (nullable)
 * @param jobs - top matching job cards (up to 3)
 * @param totalCount - total new matching jobs found (may be > 3)
 */
export async function sendJobMatchNotification(
  email: string,
  name: string | null,
  jobs: JobMatchItem[],
  totalCount: number,
): Promise<void> {
  const greeting = name ? `Hi ${name.split(' ')[0]}` : 'Hi'
  const subject =
    totalCount === 1
      ? '1 new job matches your profile on Hiretivo'
      : `${totalCount} new jobs match your profile on Hiretivo`

  const jobRows = jobs
    .map(
      (j) =>
        `<tr>
          <td style="padding:12px 0;border-bottom:1px solid #e5e7eb">
            <a href="https://hiretivo.com/jobs/${j.id}" style="font-weight:600;color:#0a0a0a;text-decoration:none">${j.title}</a>
            <span style="color:#6b7280"> at ${j.companyName}</span>
          </td>
          <td style="padding:12px 0;border-bottom:1px solid #e5e7eb;text-align:right">
            <span style="background:#7560ff;color:#fff;font-size:12px;font-weight:600;padding:2px 10px;border-radius:9999px">${j.matchScore}% match</span>
          </td>
        </tr>`,
    )
    .join('')

  const html = `
    <div style="font-family:Inter,sans-serif;max-width:560px;margin:0 auto;background:#fff;border:1px solid #e5e7eb;border-radius:12px;padding:32px">
      <p style="font-size:16px;color:#0a0a0a;margin:0 0 8px">${greeting},</p>
      <p style="font-size:16px;color:#6b7280;margin:0 0 24px">${subject}.</p>
      <table style="width:100%;border-collapse:collapse">${jobRows}</table>
      ${totalCount > jobs.length ? `<p style="margin:16px 0 0;font-size:14px;color:#6b7280">+ ${totalCount - jobs.length} more matching jobs.</p>` : ''}
      <div style="margin-top:24px">
        <a href="https://hiretivo.com/seeker/dashboard" style="background:#7560ff;color:#fff;font-size:14px;font-weight:500;padding:10px 20px;border-radius:8px;text-decoration:none;display:inline-block">View all jobs</a>
      </div>
    </div>`

  await resend.emails.send({
    from: 'Hiretivo <jobs@hiretivo.com>',
    to: email,
    subject,
    html,
  })
}
```

**Step 3 — Create cron `src/app/api/cron/job-match-notification/route.ts`:**

```typescript
import { NextRequest, NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'
import { redis } from '@/lib/redis'
import { JobStatus, VerificationStatus } from '@/constants/enums'
import { computeWeightedJobMatch } from '@/lib/match'
import { sendJobMatchNotification } from '@/lib/email/job-match-notification'
import { format } from 'date-fns'

const CRON_SECRET = process.env.CRON_SECRET ?? 'local-dev-secret'
const MATCH_THRESHOLD = 60

/*
 * GET /api/cron/job-match-notification
 * Daily cron: finds jobs posted in the last 24hrs, computes weighted match scores
 * for all VERIFIED engineers, and sends a digest to engineers with score ≥ 60.
 * Rate limit: max 1 digest per engineer per calendar day via Redis.
 * Schedule: daily at 09:00 UTC (vercel.json)
 */
export async function GET(req: NextRequest) {
  const authHeader = req.headers.get('authorization')
  if (authHeader !== `Bearer ${CRON_SECRET}`) {
    return new NextResponse('Unauthorized', { status: 401 })
  }

  const now = new Date()
  const since = new Date(now.getTime() - 24 * 60 * 60 * 1000)
  const todayKey = format(now, 'yyyy-MM-dd')

  // 1. Fetch new OPEN jobs published in last 24hrs
  const newJobs = await prisma.job.findMany({
    where: {
      status: JobStatus.OPEN,
      publishedAt: { gte: since },
    },
    select: {
      id: true,
      title: true,
      salaryMin: true,
      salaryMax: true,
      yearsExpRequired: true,
      timezoneOverlapHours: true,
      skills: { select: { skill: { select: { name: true } } } },
      recruiter: {
        select: { recruiterProfile: { select: { companyName: true } } },
      },
    },
  })

  if (newJobs.length === 0) {
    return NextResponse.json({ processed: 0, sent: 0, reason: 'no_new_jobs' })
  }

  // 2. Fetch all VERIFIED engineers with their match-relevant fields
  const engineers = await prisma.user.findMany({
    where: { verificationStatus: VerificationStatus.VERIFIED },
    select: {
      id: true,
      email: true,
      name: true,
      salaryMin: true,
      salaryMax: true,
      timezoneOverlapHours: true,
      availableFrom: true,
      seekerProfile: {
        select: {
          yearsExperience: true,
          skills: { select: { skill: { select: { name: true } } } },
        },
      },
    },
    take: 500, // hard cap per run — scale with pagination later
  })

  let processed = 0
  let sent = 0
  let skipped = 0

  for (const engineer of engineers) {
    if (!engineer.email) { skipped++; continue }

    // Rate limit: 1 digest per engineer per day
    const redisKey = `job-match-notif:${engineer.id}:${todayKey}`
    const already = await redis.get(redisKey)
    if (already) { skipped++; continue }

    const engineerSkills = engineer.seekerProfile?.skills.map(s => s.skill.name) ?? []

    // Score each new job for this engineer
    const scoredJobs = newJobs
      .map((job) => ({
        id: job.id,
        title: job.title,
        companyName: job.recruiter.recruiterProfile?.companyName ?? 'Unknown Company',
        matchScore: computeWeightedJobMatch({
          engineerSkills,
          jobRequiredSkills: job.skills.map(s => s.skill.name),
          engineerSalaryMin: engineer.salaryMin ?? null,
          engineerSalaryMax: engineer.salaryMax ?? null,
          jobSalaryMin: job.salaryMin ?? null,
          jobSalaryMax: job.salaryMax ?? null,
          engineerYearsExp: engineer.seekerProfile?.yearsExperience ?? null,
          jobYearsExpRequired: job.yearsExpRequired ?? null,
          engineerTimezoneOverlapHours: engineer.timezoneOverlapHours ?? null,
          jobTimezoneOverlapHours: job.timezoneOverlapHours ?? null,
          engineerAvailableFrom: engineer.availableFrom ?? null,
        }),
      }))
      .filter(j => j.matchScore >= MATCH_THRESHOLD)
      .sort((a, b) => b.matchScore - a.matchScore)

    if (scoredJobs.length === 0) { processed++; continue }

    try {
      await sendJobMatchNotification(
        engineer.email,
        engineer.name,
        scoredJobs.slice(0, 3),
        scoredJobs.length,
      )
      // Mark as sent for today — TTL 25hrs to avoid edge-of-day gaps
      await redis.set(redisKey, '1', { ex: 25 * 60 * 60 })
      sent++
    } catch (err) {
      console.error(`[job-match-notif] Failed for engineer ${engineer.id}:`, err)
    }
    processed++
  }

  return NextResponse.json({ processed, sent, skipped, timestamp: now.toISOString() })
}
```
