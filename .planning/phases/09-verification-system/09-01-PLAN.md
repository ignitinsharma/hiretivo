model VerificationReview {
  id           String             @id @default(cuid())
  userId       String
  reviewedBy   String?
  status       VerificationStatus
  scoreGiven   Int?
  notes        Json?
  aiSummary    Json?
  createdAt    DateTime           @default(now())
  user         User               @relation(fields: [userId], references: [id], onDelete: Cascade)
  @@index([userId])
  @@index([status])
  @@index([createdAt])
}

model EngineerUnlock {
  id           String   @id @default(cuid())
  recruiterId  String
  engineerId   String
  source       String
  unlockedAt   DateTime @default(now())
  recruiter    User     @relation("RecruiterUnlocks", fields: [recruiterId], references: [id], onDelete: Cascade)
  engineer     User     @relation("EngineerUnlocks", fields: [engineerId], references: [id], onDelete: Cascade)
  @@unique([recruiterId, engineerId])
  @@index([recruiterId])
  @@index([recruiterId, unlockedAt])
}

model EmployerFeedback {
  id                String   @id @default(cuid())
  engineerId        String
  recruiterId       String
  jobId             String?
  feedbackRequestId String   @unique
  rating            Int
  comment           String?  @db.Text
  createdAt         DateTime @default(now())
  engineer          User     @relation("FeedbackReceived", fields: [engineerId], references: [id], onDelete: Cascade)
  recruiter         User     @relation("FeedbackGiven", fields: [recruiterId], references: [id], onDelete: Cascade)
  @@index([engineerId])
}

model SavedEngineer {
  id           String   @id @default(cuid())
  recruiterId  String
  engineerId   String
  createdAt    DateTime @default(now())
  recruiter    User     @relation("SavedByRecruiter", fields: [recruiterId], references: [id], onDelete: Cascade)
  engineer     User     @relation("SavedEngineer", fields: [engineerId], references: [id], onDelete: Cascade)
  @@unique([recruiterId, engineerId])
  @@index([recruiterId])
}

model BlockedRecruiter {
  id              String   @id @default(cuid())
  engineerId      String
  recruiterDomain String
  createdAt       DateTime @default(now())
  engineer        User     @relation(fields: [engineerId], references: [id], onDelete: Cascade)
  @@unique([engineerId, recruiterDomain])
  @@index([engineerId])
}

model FeedbackRequest {
  id          String    @id @default(cuid())
  tokenHash   String    @unique
  engineerId  String
  recruiterId String
  jobId       String?
  expiresAt   DateTime
  usedAt      DateTime?
  createdAt   DateTime  @default(now())
  engineer    User      @relation("FeedbackRequests", fields: [engineerId], references: [id], onDelete: Cascade)
  @@index([engineerId])
  @@index([tokenHash])
}

export const DODO_PRODUCT_IDS = {
  STARTER_MONTHLY_USD: process.env.DODO_STARTER_MONTHLY_PRODUCT_ID ?? "",
  STARTER_ANNUAL_USD:  process.env.DODO_STARTER_ANNUAL_PRODUCT_ID ?? "",

  GROWTH_ANNUAL_USD:   process.env.DODO_GROWTH_ANNUAL_PRODUCT_ID ?? "",
  SCALE_MONTHLY_USD:   process.env.DODO_SCALE_MONTHLY_PRODUCT_ID ?? "",
} as const;

export const DODO_PRODUCT_TO_PLAN: Record<string, PlanTier> = {
  [DODO_PRODUCT_IDS.STARTER_MONTHLY_USD]: PlanTier.STARTER,
  [DODO_PRODUCT_IDS.STARTER_ANNUAL_USD]:  PlanTier.STARTER,
  [DODO_PRODUCT_IDS.STARTER_MONTHLY_INR]: PlanTier.STARTER,
  [DODO_PRODUCT_IDS.STARTER_ANNUAL_INR]:  PlanTier.STARTER,
  [DODO_PRODUCT_IDS.GROWTH_MONTHLY_USD]:  PlanTier.GROWTH,
  [DODO_PRODUCT_IDS.GROWTH_ANNUAL_USD]:   PlanTier.GROWTH,
  [DODO_PRODUCT_IDS.SCALE_MONTHLY_USD]:   PlanTier.SCALE,
};

/*
 * getDodoProductId
 * Maps plan tier + billing period + currency to the Dodo Payments product ID.
 * Returns null when env var is unset (graceful degradation in dev/staging).
 */
export function getDodoProductId(
  tier: PlanTier.STARTER | PlanTier.GROWTH | PlanTier.SCALE,
  period: BillingPeriod,
  currency: CurrencyCode.USD | CurrencyCode.INR = CurrencyCode.USD,
): string | null {
  if (tier === PlanTier.STARTER) {
    if (currency === CurrencyCode.INR) {
      return period === BillingPeriod.ANNUAL ? DODO_PRODUCT_IDS.STARTER_ANNUAL_INR || null : DODO_PRODUCT_IDS.STARTER_MONTHLY_INR || null;
    }
    return period === BillingPeriod.ANNUAL ? DODO_PRODUCT_IDS.STARTER_ANNUAL_USD || null : DODO_PRODUCT_IDS.STARTER_MONTHLY_USD || null;
  }
  if (tier === PlanTier.GROWTH) {
    return period === BillingPeriod.ANNUAL ? DODO_PRODUCT_IDS.GROWTH_ANNUAL_USD || null : DODO_PRODUCT_IDS.GROWTH_MONTHLY_USD || null;
  }
  return DODO_PRODUCT_IDS.SCALE_MONTHLY_USD || null;
}
