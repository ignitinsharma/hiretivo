/*
 * GET /api/engineer/verification
 * Returns the authenticated engineer's current verificationStatus and last VerificationReview.
 * Rate-limited: 30 req/min per user (read).
 *
 * POST /api/engineer/verification
 * Submits Phase B verification. Validates all fields, enforces reapply gate,
 * reserves profileSlug, sets verificationStatus=PENDING, emails admin.
 * Rate-limited: 5 req/min per user (write).
 * Body: { githubUrl, legalName, bio, topProjects: [{name, description, url}], videoIntroUrl?, profileSlug }
 */
import { NextRequest, NextResponse } from "next/server";
import { auth } from "@/auth";
import { UserRole, VerificationStatus } from "@/constants/enums";
import { prisma } from "@/lib/prisma";
import { dataFetchRateLimiter, writeRateLimiter, getRateLimitIp } from "@/lib/rate-limit";
import { sendEmail } from "@/lib/resend";
import { isAfter } from "date-fns";
import { z } from "zod";

const RESERVED_SLUGS = new Set([
  "admin","api","pricing","jobs","engineers","recruiter","seeker","login","signup",
  "dashboard","health","feedback","e","app","www","support","help","about","blog",
  "terms","privacy","careers","hire","hiretivo"
]);

const projectSchema = z.object({
  name: z.string().min(1).max(100),
  description: z.string().min(1).max(500),
  url: z.string().url().startsWith("https://"),
});

const verificationSubmitSchema = z.object({
  githubUrl: z.string().regex(/^https:\/\/github\.com\/.+/, "Must be a valid GitHub profile URL"),
  legalName: z.string().min(2).max(100),
  bio: z.string().min(1).max(5000),
  topProjects: z.array(projectSchema).min(1).max(3),
  videoIntroUrl: z.string().url().startsWith("https://").optional().nullable(),
  profileSlug: z.string()
    .min(3).max(30)
    .regex(/^[a-z0-9][a-z0-9-]*[a-z0-9]$/, "Slug must be lowercase alphanumeric with hyphens; no leading/trailing hyphens"),
});

export async function GET(req: NextRequest) {
  const session = await auth();
  if (!session?.user?.id) return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
  if (session.user.role !== UserRole.SEEKER) return NextResponse.json({ error: "Forbidden" }, { status: 403 });

  const { success } = await dataFetchRateLimiter.limit(session.user.id);
  if (!success) return NextResponse.json({ error: "Too many requests" }, { status: 429 });

  const user = await prisma.user.findUnique({
    where: { id: session.user.id },
    select: {
      verificationStatus: true,
      hiretivoScore: true,
      reapplyAfter: true,
      profileSlug: true,
      verificationReviews: {
        orderBy: { createdAt: "desc" },
        take: 1,
        select: { status: true, scoreGiven: true, notes: true, createdAt: true },
      },
      seekerProfile: { select: { profilePublic: true, profilePublicConsentAt: true } },
    },
  });
  if (!user) return NextResponse.json({ error: "Not found" }, { status: 404 });

  return NextResponse.json({
    verificationStatus: user.verificationStatus,
    hiretivoScore: user.hiretivoScore,
    reapplyAfter: user.reapplyAfter,
    profileSlug: user.profileSlug,
    profilePublic: user.seekerProfile?.profilePublic ?? false,
    lastReview: user.verificationReviews[0] ?? null,
  });
}

export async function POST(req: NextRequest) {
  const ip = getRateLimitIp(req);
  const session = await auth();
  if (!session?.user?.id) return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
  if (session.user.role !== UserRole.SEEKER) return NextResponse.json({ error: "Forbidden" }, { status: 403 });

  const { success } = await writeRateLimiter.limit(session.user.id);
  if (!success) return NextResponse.json({ error: "Too many requests" }, { status: 429 });

  let body: unknown;
  try { body = await req.json(); } catch { return NextResponse.json({ error: "Invalid JSON" }, { status: 400 }); }

  const parsed = verificationSubmitSchema.safeParse(body);
  if (!parsed.success) return NextResponse.json({ error: parsed.error.issues[0]?.message ?? "Validation failed" }, { status: 400 });

  const { githubUrl, legalName, bio, topProjects, videoIntroUrl, profileSlug } = parsed.data;

  // Server-side reserved word check
  if (RESERVED_SLUGS.has(profileSlug.toLowerCase())) {
    return NextResponse.json({ error: "That slug is reserved. Please choose a different one." }, { status: 400 });
  }

  const user = await prisma.user.findUnique({
    where: { id: session.user.id },
    select: { verificationStatus: true, reapplyAfter: true },
  });
  if (!user) return NextResponse.json({ error: "Not found" }, { status: 404 });

  // Reapply gate
  if (user.reapplyAfter && isAfter(user.reapplyAfter, new Date())) {
    return NextResponse.json({ error: "Too soon to reapply", reapplyAfter: user.reapplyAfter }, { status: 429 });
  }

  // Only NOT_SUBMITTED and REJECTED can submit
  const canSubmit = [VerificationStatus.NOT_SUBMITTED, VerificationStatus.REJECTED].includes(user.verificationStatus);
  if (!canSubmit) {
    return NextResponse.json({ error: "Verification already in progress or completed" }, { status: 409 });
  }

  // Check slug availability (excluding current user)
  const slugTaken = await prisma.user.findFirst({
    where: { profileSlug, NOT: { id: session.user.id } },
    select: { id: true },
  });
  if (slugTaken) return NextResponse.json({ error: "Slug already taken" }, { status: 409 });

  const githubUsername = githubUrl.replace("https://github.com/", "").split("/")[0];

  await prisma.$transaction([
    prisma.user.update({
      where: { id: session.user.id },
      data: {
        verificationStatus: VerificationStatus.PENDING,
        profileSlug,
        reapplyAfter: null,
      },
    }),
    prisma.seekerProfile.update({
      where: { userId: session.user.id },
      data: {
        githubUsername,
        bio,
        videoIntroUrl: videoIntroUrl ?? null,
        topProjects,
        verificationNotes: null,
        verificationReviewedAt: null,
        verificationReviewedBy: null,
      },
    }),
    prisma.verificationReview.create({
      data: {
        userId: session.user.id,
        status: VerificationStatus.PENDING,
      },
    }),
  ]);

  // Email admin
  const adminEmail = process.env.ADMIN_EMAIL ?? "admin@hiretivo.com";
  await sendEmail({
    from: "Hiretivo <noreply@hiretivo.com>",
    to: adminEmail,
    subject: `New verification submission: ${legalName} (@${githubUsername})`,
    html: `<p>New engineer verification submitted.</p>
      <p><strong>Name:</strong> ${legalName}<br>
      <strong>GitHub:</strong> <a href="${githubUrl}">${githubUrl}</a><br>
      <strong>Slug:</strong> hiretivo.com/e/${profileSlug}</p>
      <p><a href="${process.env.NEXT_PUBLIC_APP_URL ?? "https://hiretivo.com"}/admin/verification">Review in admin queue →</a></p>`,
  }).catch((err) => console.error("[verification POST] Admin email failed (non-fatal):", err));

  return NextResponse.json({ success: true, message: "Under review — we'll email you within 48 hours." }, { status: 201 });
}

/*
 * GET /api/engineer/check-slug?slug=priyasharma
 * Returns { available: boolean, suggestion?: string }.
 * Checks uniqueness against User.profileSlug; blocks reserved words.
 * Suggestion algo: append "2", "3", "4"... until available.
 * Rate-limited: 30 req/min per IP (debounce-friendly read).
 * PIVS-14.
 */
import { NextRequest, NextResponse } from "next/server";
import { dataFetchRateLimiter, getRateLimitIp } from "@/lib/rate-limit";
import { prisma } from "@/lib/prisma";
import { auth } from "@/auth";

const RESERVED_SLUGS = new Set([
  "admin","api","pricing","jobs","engineers","recruiter","seeker","login","signup",
  "dashboard","health","feedback","e","app","www","support","help","about","blog",
  "terms","privacy","careers","hire","hiretivo"
]);

const SLUG_REGEX = /^[a-z0-9][a-z0-9-]*[a-z0-9]$/;

export async function GET(req: NextRequest) {
  const ip = getRateLimitIp(req);
  const { success } = await dataFetchRateLimiter.limit(ip);
  if (!success) return NextResponse.json({ error: "Too many requests" }, { status: 429 });

  const session = await auth();
  if (!session?.user?.id) return NextResponse.json({ error: "Unauthorized" }, { status: 401 });

  const slug = req.nextUrl.searchParams.get("slug")?.toLowerCase() ?? "";

  if (!slug || slug.length < 3 || slug.length > 30 || !SLUG_REGEX.test(slug)) {
    return NextResponse.json({ available: false }, { status: 200 });
  }

  if (RESERVED_SLUGS.has(slug)) {
    const suggestion = slug + "2";
    return NextResponse.json({ available: false, suggestion }, { status: 200 });
  }

  const existing = await prisma.user.findFirst({
    where: { profileSlug: slug, NOT: { id: session.user.id } },
    select: { id: true },
  });

  if (!existing) return NextResponse.json({ available: true }, { status: 200 });

  // Find available suggestion
  let suffix = 2;
  let suggestion: string | undefined;
  while (suffix <= 99) {
    const candidate = slug + suffix;
    const taken = await prisma.user.findFirst({ where: { profileSlug: candidate }, select: { id: true } });
    if (!taken) { suggestion = candidate; break; }
    suffix++;
  }

  return NextResponse.json({ available: false, suggestion }, { status: 200 });
}

/*
 * POST /api/engineer/github-scan
 * Phase 9 stub: validates GitHub profile URL, checks if public, counts public repos.
 * Returns { status: "success" | "private" | "not_found" | "timeout", publicRepoCount?: number }.
 * AI analysis deferred to Phase 12 — this is a lightweight HTTP check only.
 * Rate-limited: 5 req/min per user (write — external request initiated).
 * PIVS-04 (scan spinner states in UI depend on this endpoint existing).
 */
import { NextRequest, NextResponse } from "next/server";
import { auth } from "@/auth";
import { UserRole } from "@/constants/enums";
import { writeRateLimiter } from "@/lib/rate-limit";
import { z } from "zod";

const githubScanSchema = z.object({
  githubUrl: z.string().regex(/^https:\/\/github\.com\/.+/, "Must be a valid GitHub profile URL"),
});

export async function POST(req: NextRequest) {
  const session = await auth();
  if (!session?.user?.id) return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
  if (session.user.role !== UserRole.SEEKER) return NextResponse.json({ error: "Forbidden" }, { status: 403 });

  const { success } = await writeRateLimiter.limit(session.user.id);
  if (!success) return NextResponse.json({ error: "Too many requests" }, { status: 429 });

  let body: unknown;
  try { body = await req.json(); } catch { return NextResponse.json({ error: "Invalid JSON" }, { status: 400 }); }

  const parsed = githubScanSchema.safeParse(body);
  if (!parsed.success) return NextResponse.json({ error: "Invalid GitHub URL" }, { status: 400 });

  const username = parsed.data.githubUrl.replace("https://github.com/", "").split("/")[0];
  if (!username) return NextResponse.json({ error: "Could not extract username from URL" }, { status: 400 });

  // Phase 9: lightweight check via GitHub API (unauthenticated — rate limit 60/hr)
  // Phase 12 will add AI analysis (Claude Haiku via OpenRouter)
  try {
    const controller = new AbortController();
    const timeout = setTimeout(() => controller.abort(), 10000); // 10s timeout

    const apiRes = await fetch(`https://api.github.com/users/${username}`, {
      headers: { "Accept": "application/vnd.github+json" },
      signal: controller.signal,
    });
    clearTimeout(timeout);

    if (apiRes.status === 404) return NextResponse.json({ status: "not_found" });
    if (!apiRes.ok) return NextResponse.json({ status: "timeout" });

    const data = await apiRes.json() as { public_repos?: number; type?: string };
    if (data.type === "Organization") return NextResponse.json({ status: "not_found" });

    const publicRepoCount = data.public_repos ?? 0;
    if (publicRepoCount === 0) return NextResponse.json({ status: "private", publicRepoCount: 0 });

    return NextResponse.json({ status: "success", publicRepoCount });
  } catch (err: unknown) {
    if (err instanceof Error && err.name === "AbortError") {
      return NextResponse.json({ status: "timeout" });
    }
    return NextResponse.json({ status: "timeout" });
  }
}

/*
 * POST /api/engineer/profile/public-consent
 * Sets SeekerProfile.profilePublic=true and profilePublicConsentAt=now().
 * Engineer must be VERIFIED to consent. Idempotent (re-consent allowed).
 * Rate-limited: 10 req/min per user (write).
 * PIVS-15.
 */
import { NextResponse } from "next/server";
import { auth } from "@/auth";
import { UserRole, VerificationStatus } from "@/constants/enums";
import { prisma } from "@/lib/prisma";
import { writeRateLimiter } from "@/lib/rate-limit";

export async function POST() {
  const session = await auth();
  if (!session?.user?.id) return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
  if (session.user.role !== UserRole.SEEKER) return NextResponse.json({ error: "Forbidden" }, { status: 403 });

  const { success } = await writeRateLimiter.limit(session.user.id);
  if (!success) return NextResponse.json({ error: "Too many requests" }, { status: 429 });

  const user = await prisma.user.findUnique({
    where: { id: session.user.id },
    select: { verificationStatus: true },
  });
  if (!user) return NextResponse.json({ error: "Not found" }, { status: 404 });
  if (user.verificationStatus !== VerificationStatus.VERIFIED) {
    return NextResponse.json({ error: "Only VERIFIED engineers can make their profile public" }, { status: 403 });
  }

  await prisma.seekerProfile.update({
    where: { userId: session.user.id },
    data: { profilePublic: true, profilePublicConsentAt: new Date() },
  });

  return NextResponse.json({ success: true });
}
