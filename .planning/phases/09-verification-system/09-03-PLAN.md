/*
 * GET /api/admin/verification/queue
 * Returns PENDING and IN_REVIEW verification submissions sorted by createdAt ASC (FIFO).
 * Lazy stale-check: any IN_REVIEW item older than 2hrs is reverted to PENDING before returning.
 * Rate-limited: 30 req/min per admin.
 * PIVS-06.
 */
import { NextResponse } from "next/server";
import { auth } from "@/auth";
import { VerificationStatus } from "@/constants/enums";
import { prisma } from "@/lib/prisma";
import { dataFetchRateLimiter } from "@/lib/rate-limit";
import { subHours } from "date-fns";

export async function GET() {
  const session = await auth();
  const isAdmin = session?.user?.email?.trim().toLowerCase() === process.env.ADMIN_EMAIL?.trim().toLowerCase();
  if (!session?.user?.id || !isAdmin) return NextResponse.json({ error: "Forbidden" }, { status: 403 });

  const { success } = await dataFetchRateLimiter.limit(session.user.id);
  if (!success) return NextResponse.json({ error: "Too many requests" }, { status: 429 });

  // Lazy stale revert: IN_REVIEW items older than 2hrs go back to PENDING
  const staleThreshold = subHours(new Date(), 2);
  await prisma.user.updateMany({
    where: {
      verificationStatus: VerificationStatus.IN_REVIEW,
      updatedAt: { lt: staleThreshold },
    },
    data: { verificationStatus: VerificationStatus.PENDING },
  });

  const submissions = await prisma.user.findMany({
    where: {
      verificationStatus: { in: [VerificationStatus.PENDING, VerificationStatus.IN_REVIEW] },
    },
    select: {
      id: true,
      name: true,
      email: true,
      verificationStatus: true,
      profileSlug: true,
      updatedAt: true,
      createdAt: true,
      seekerProfile: {
        select: {
          headline: true,
          primaryRole: true,
          githubUsername: true,
          salaryMinUsd: true,
          salaryMaxUsd: true,
          topProjects: true,
        },
      },
      verificationReviews: {
        orderBy: { createdAt: "desc" },
        take: 1,
        select: { createdAt: true, status: true },
      },
    },
    orderBy: { createdAt: "asc" }, // FIFO
  });

  return NextResponse.json({ submissions });
}

--- src/app/api/admin/verification/[id]/route.ts ---
/*
 * GET /api/admin/verification/[id]
 * Returns full engineer submission detail.
 * Side effect: sets verificationStatus=IN_REVIEW when currently PENDING (admin opened it).
 * If already IN_REVIEW by another admin, returns a "stale" warning flag.
 * Rate-limited: 30 req/min per admin.
 * PIVS-06 (auto IN_REVIEW on open).
 */
import { NextRequest, NextResponse } from "next/server";
import { auth } from "@/auth";
import { VerificationStatus } from "@/constants/enums";
import { prisma } from "@/lib/prisma";
import { dataFetchRateLimiter } from "@/lib/rate-limit";

export async function GET(req: NextRequest, { params }: { params: { id: string } }) {
  const session = await auth();
  const isAdmin = session?.user?.email?.trim().toLowerCase() === process.env.ADMIN_EMAIL?.trim().toLowerCase();
  if (!session?.user?.id || !isAdmin) return NextResponse.json({ error: "Forbidden" }, { status: 403 });

  const { success } = await dataFetchRateLimiter.limit(session.user.id);
  if (!success) return NextResponse.json({ error: "Too many requests" }, { status: 429 });

  const engineer = await prisma.user.findUnique({
    where: { id: params.id },
    select: {
      id: true, name: true, email: true, createdAt: true, updatedAt: true,
      verificationStatus: true, hiretivoScore: true, reapplyAfter: true, profileSlug: true,
      timezoneOverlapHours: true, availableFrom: true,
      seekerProfile: {
        select: {
          headline: true, bio: true, primaryRole: true, location: true, yearsExperience: true,
          githubUsername: true, githubScore: true, githubAnalysis: true, githubScannedAt: true,
          videoIntroUrl: true, topProjects: true,
          salaryMinUsd: true, salaryMaxUsd: true,
          verificationNotes: true, verificationReviewedAt: true, verificationReviewedBy: true,
          skills: { select: { skill: { select: { name: true } }, level: true } },
          workExperiences: { select: { company: true, title: true, startDate: true, endDate: true, isCurrent: true } },
        },
      },
      verificationReviews: {
        orderBy: { createdAt: "desc" },
        select: { id: true, status: true, scoreGiven: true, notes: true, createdAt: true, reviewedBy: true },
      },
    },
  });

  if (!engineer) return NextResponse.json({ error: "Not found" }, { status: 404 });

  let alreadyInReview = false;
  if (engineer.verificationStatus === VerificationStatus.IN_REVIEW) {
    alreadyInReview = true; // Could show "Another admin has this open" banner in UI
  }

  // Auto-transition PENDING → IN_REVIEW on admin open
  if (engineer.verificationStatus === VerificationStatus.PENDING) {
    await prisma.user.update({
      where: { id: params.id },
      data: { verificationStatus: VerificationStatus.IN_REVIEW },
    });
  }

  return NextResponse.json({ engineer, alreadyInReview });
}

--- src/app/api/admin/verification/[id]/review/route.ts ---
/*
 * POST /api/admin/verification/[id]/review
 * Approves or rejects a verification submission.
 * APPROVED: verificationStatus=VERIFIED, hiretivoScore=scoreGiven, approval email sent.
 * REJECTED: verificationStatus=REJECTED, reapplyAfter set, notes stored, rejection email sent.
 * Body: { decision: "APPROVED" | "REJECTED", score?: number, reasons?: string[], customNote?: string, reapplyDays?: 30 | 60 | 90 }
 * Rate-limited: 5 req/min per admin (write).
 * PIVS-07, PIVS-08.
 */
import { NextRequest, NextResponse } from "next/server";
import { auth } from "@/auth";
import { VerificationStatus } from "@/constants/enums";
import { prisma } from "@/lib/prisma";
import { writeRateLimiter } from "@/lib/rate-limit";
import { sendApprovalEmail } from "@/lib/emails/verification-approval";
import { sendRejectionEmail } from "@/lib/emails/verification-rejection";
import { addDays } from "date-fns";
import { z } from "zod";

const reviewSchema = z.discriminatedUnion("decision", [
  z.object({
    decision: z.literal("APPROVED"),
    score: z.number().int().min(0).max(100),
  }),
  z.object({
    decision: z.literal("REJECTED"),
    reasons: z.array(z.string()).min(1, "Select at least one rejection reason"),
    customNote: z.string().max(1000).optional(),
    reapplyDays: z.union([z.literal(30), z.literal(60), z.literal(90)]),
  }),
]);

export async function POST(req: NextRequest, { params }: { params: { id: string } }) {
  const session = await auth();
  const isAdmin = session?.user?.email?.trim().toLowerCase() === process.env.ADMIN_EMAIL?.trim().toLowerCase();
  if (!session?.user?.id || !isAdmin) return NextResponse.json({ error: "Forbidden" }, { status: 403 });

  const { success } = await writeRateLimiter.limit(session.user.id);
  if (!success) return NextResponse.json({ error: "Too many requests" }, { status: 429 });

  let body: unknown;
  try { body = await req.json(); } catch { return NextResponse.json({ error: "Invalid JSON" }, { status: 400 }); }

  const parsed = reviewSchema.safeParse(body);
  if (!parsed.success) return NextResponse.json({ error: parsed.error.issues[0]?.message ?? "Validation failed" }, { status: 400 });

  const engineer = await prisma.user.findUnique({
    where: { id: params.id },
    select: { id: true, name: true, email: true, verificationStatus: true, profileSlug: true },
  });
  if (!engineer) return NextResponse.json({ error: "Not found" }, { status: 404 });
  if (engineer.verificationStatus !== VerificationStatus.IN_REVIEW) {
    return NextResponse.json({ error: "Engineer is not IN_REVIEW — cannot review" }, { status: 409 });
  }

  const adminEmail = session.user.email!;

  if (parsed.data.decision === "APPROVED") {
    await prisma.$transaction([
      prisma.user.update({
        where: { id: params.id },
        data: {
          verificationStatus: VerificationStatus.VERIFIED,
          hiretivoScore: parsed.data.score,
          scoreUpdatedAt: new Date(),
        },
      }),
      prisma.seekerProfile.update({
        where: { userId: params.id },
        data: {
          verificationReviewedAt: new Date(),
          verificationReviewedBy: adminEmail,
        },
      }),
      prisma.verificationReview.create({
        data: {
          userId: params.id,
          reviewedBy: adminEmail,
          status: VerificationStatus.VERIFIED,
          scoreGiven: parsed.data.score,
        },
      }),
    ]);

    await sendApprovalEmail({
      to: engineer.email!,
      name: engineer.name ?? "Engineer",
      score: parsed.data.score,
      profileSlug: engineer.profileSlug ?? "",
    }).catch((err) => console.error("[review POST] Approval email failed (non-fatal):", err));

  } else {
    const reapplyAfter = addDays(new Date(), parsed.data.reapplyDays);

    await prisma.$transaction([
      prisma.user.update({
        where: { id: params.id },
        data: {
          verificationStatus: VerificationStatus.REJECTED,
          reapplyAfter,
        },
      }),
      prisma.seekerProfile.update({
        where: { userId: params.id },
        data: {
          verificationNotes: parsed.data.customNote ?? null,
          verificationReviewedAt: new Date(),
          verificationReviewedBy: adminEmail,
        },
      }),
      prisma.verificationReview.create({
        data: {
          userId: params.id,
          reviewedBy: adminEmail,
          status: VerificationStatus.REJECTED,
          notes: {
            reasons: parsed.data.reasons,
            customNote: parsed.data.customNote ?? "",
            reapplyDays: parsed.data.reapplyDays,
          },
        },
      }),
    ]);

    await sendRejectionEmail({
      to: engineer.email!,
      name: engineer.name ?? "Engineer",
      reasons: parsed.data.reasons,
      customNote: parsed.data.customNote,
      reapplyAfter,
    }).catch((err) => console.error("[review POST] Rejection email failed (non-fatal):", err));
  }

  return NextResponse.json({ success: true });
}

/*
 * GET /api/cron/verification-stale
 * Vercel cron job (every 30 minutes): reverts IN_REVIEW items older than 2hrs to PENDING.
 * Protected by CRON_SECRET header to prevent public triggering.
 * PIVS-06.
 */
import { NextRequest, NextResponse } from "next/server";
import { VerificationStatus } from "@/constants/enums";
import { prisma } from "@/lib/prisma";
import { subHours } from "date-fns";

export async function GET(req: NextRequest) {
  const cronSecret = req.headers.get("x-cron-secret");
  if (!cronSecret || cronSecret !== process.env.CRON_SECRET) {
    return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
  }

  const staleThreshold = subHours(new Date(), 2);

  const result = await prisma.user.updateMany({
    where: {
      verificationStatus: VerificationStatus.IN_REVIEW,
      updatedAt: { lt: staleThreshold },
    },
