/*
 * /admin/verification
 * Server component: fetches PENDING+IN_REVIEW queue via internal API call.
 * Middleware already gates /admin/* on ADMIN_EMAIL — this page trusts that.
 */
import { VerificationQueue } from "@/modules/admin-verification/components/VerificationQueue";
import { prisma } from "@/lib/prisma";
import { VerificationStatus } from "@/constants/enums";
import { subHours } from "date-fns";

export const revalidate = 0; // Always fresh

export default async function AdminVerificationPage() {
  // Lazy stale revert (mirrors queue API lazy check)
  const staleThreshold = subHours(new Date(), 2);
  await prisma.user.updateMany({
    where: { verificationStatus: VerificationStatus.IN_REVIEW, updatedAt: { lt: staleThreshold } },
    data: { verificationStatus: VerificationStatus.PENDING },
  });

  const submissions = await prisma.user.findMany({
    where: { verificationStatus: { in: [VerificationStatus.PENDING, VerificationStatus.IN_REVIEW] } },
    select: {
      id: true, name: true, email: true,
      verificationStatus: true, profileSlug: true, updatedAt: true, createdAt: true,
      seekerProfile: { select: { headline: true, primaryRole: true, githubUsername: true, salaryMinUsd: true, salaryMaxUsd: true } },
      verificationReviews: { orderBy: { createdAt: "desc" }, take: 1, select: { createdAt: true } },
    },
    orderBy: { createdAt: "asc" },
  });

  return (
    <main className="max-w-4xl mx-auto px-4 py-8">
      <div className="mb-6">
        <h1 className="font-bold text-2xl text-brand-foreground tracking-tight">Verification Queue</h1>
        <p className="text-brand-foreground-muted text-sm mt-1">{submissions.length} submission{submissions.length !== 1 ? "s" : ""} pending review</p>
      </div>
      <VerificationQueue submissions={submissions} />
    </main>
  );
}

/*
 * VerificationQueue
 * Renders the list of submissions. Empty state if none.
 * @param submissions - Array of submission objects from /admin/verification page
 */
import { VerificationQueueRow } from "./VerificationQueueRow";
import type { VerificationStatus } from "@/constants/enums";

interface Submission {
  id: string; name: string | null; email: string | null;
  verificationStatus: VerificationStatus; profileSlug: string | null;
  updatedAt: Date; createdAt: Date;
  seekerProfile: { headline: string | null; primaryRole: string | null; githubUsername: string | null; salaryMinUsd: number | null; salaryMaxUsd: number | null; } | null;
  verificationReviews: { createdAt: Date }[];
}

interface Props { submissions: Submission[]; }

export function VerificationQueue({ submissions }: Props) {
  if (submissions.length === 0) {
    return (
      <div className="rounded-xl border border-brand-border bg-white p-12 text-center">
        <p className="text-brand-foreground font-semibold mb-1">Queue is clear</p>
        <p className="text-brand-foreground-muted text-sm">No pending or in-review submissions.</p>
      </div>
    );
  }
  return (
    <div className="space-y-3">
      {submissions.map((s) => <VerificationQueueRow key={s.id} submission={s} />)}
    </div>
  );
}

/*
 * VerificationQueueRow
 * Single row in admin queue. Shows name, role, GitHub, status badge, submitted date.
 * IN_REVIEW >2hrs shows stale warning. Clicking opens /admin/verification/[id].
 */
"use client";
import { format, differenceInHours } from "date-fns";
import { VerificationStatus } from "@/constants/enums";
import type { ComponentProps } from "react";

type Submission = ComponentProps<typeof import("./VerificationQueue").VerificationQueue>["submissions"][number];

interface Props { submission: Submission; }

export function VerificationQueueRow({ submission: s }: Props) {
  const isStale = s.verificationStatus === VerificationStatus.IN_REVIEW
    && differenceInHours(new Date(), new Date(s.updatedAt)) >= 2;

  const salaryDisplay = s.seekerProfile?.salaryMinUsd && s.seekerProfile?.salaryMaxUsd
    ? `$${Math.round(s.seekerProfile.salaryMinUsd / 1000)}k–$${Math.round(s.seekerProfile.salaryMaxUsd / 1000)}k`
    : null;

  return (
    <a href={`/admin/verification/${s.id}`}
      className="block rounded-xl border border-brand-border bg-white p-5 hover:border-[#d1d5db] transition-colors">
      <div className="flex flex-col sm:flex-row sm:items-start sm:justify-between gap-3">
        <div className="min-w-0">
          <div className="flex items-center gap-2 flex-wrap mb-1">
            <span className="font-semibold text-brand-foreground">{s.name ?? "—"}</span>
            <span className={`text-xs font-medium px-2 py-0.5 rounded-full border ${
              s.verificationStatus === VerificationStatus.IN_REVIEW
                ? "bg-yellow-50 text-yellow-700 border-yellow-200"
                : "bg-brand-surface text-brand-foreground-muted border-brand-border"
            }`}>
              {s.verificationStatus === VerificationStatus.IN_REVIEW ? "In Review" : "Pending"}
            </span>
            {isStale && (
              <span className="text-xs text-orange-600 font-medium">⚠ Stale — reverting to Pending</span>
            )}
          </div>
          <p className="text-sm text-brand-foreground-muted truncate">{s.email}</p>
          {s.seekerProfile?.primaryRole && (
            <p className="text-sm text-brand-foreground-muted">{s.seekerProfile.primaryRole}{salaryDisplay ? ` · ${salaryDisplay}` : ""}</p>
          )}
          {s.seekerProfile?.githubUsername && (
            <p className="text-xs text-brand-foreground-secondary mt-1">github.com/{s.seekerProfile.githubUsername}</p>
          )}
        </div>
        <div className="text-xs text-brand-foreground-secondary shrink-0">
          Submitted {format(new Date(s.createdAt), "MMM d, yyyy")}
        </div>
      </div>
    </a>
  );
}

/*
 * AdminReviewPanel
 * Client component: toggles between ApproveForm and RejectForm.
 * Shows "Already decided" state when canReview=false.
 * @param engineerId    - User ID being reviewed
 * @param canReview     - False if status is not IN_REVIEW/PENDING
 * @param currentStatus - Current VerificationStatus for display
 */
"use client";
import { useState } from "react";
import { VerificationStatus } from "@/constants/enums";
import { ApproveForm } from "./ApproveForm";
import { RejectForm } from "./RejectForm";

interface Props {
  engineerId: string;
  canReview: boolean;
  currentStatus: VerificationStatus;
}

export function AdminReviewPanel({ engineerId, canReview, currentStatus }: Props) {
  const [activeForm, setActiveForm] = useState<"approve" | "reject" | null>(null);
  const [done, setDone] = useState(false);

  if (!canReview) {
    return (
      <div className="rounded-xl border border-brand-border bg-white p-6">
        <p className="text-sm font-medium text-brand-foreground mb-1">Review complete</p>
        <p className="text-sm text-brand-foreground-muted">Status: <span className="font-medium">{currentStatus}</span></p>
      </div>
    );
  }

  if (done) {
    return (
      <div className="rounded-xl border border-brand-border bg-white p-6">
        <p className="text-brand-success font-semibold text-sm">Decision submitted ✓</p>
        <p className="text-brand-foreground-muted text-sm mt-1">Engineer has been notified by email.</p>
        <a href="/admin/verification" className="text-sm text-brand-primary hover:underline mt-3 block">← Back to queue</a>
      </div>
    );
  }

  return (
    <div className="rounded-xl border border-brand-border bg-white p-6 space-y-4">
      <p className="text-sm font-semibold text-brand-foreground">Review Decision</p>

      {!activeForm && (
        <div className="flex flex-col gap-2">
          <button onClick={() => setActiveForm("approve")}
            className="bg-brand-primary text-white font-medium text-sm px-4 py-2.5 rounded-lg hover:bg-[#5a47d9] transition-colors">
            Approve →
          </button>
          <button onClick={() => setActiveForm("reject")}
            className="bg-white text-brand-danger font-medium text-sm px-4 py-2.5 rounded-lg border border-brand-danger hover:bg-red-50 transition-colors">
            Reject
          </button>
        </div>
      )}

      {activeForm === "approve" && (
        <ApproveForm engineerId={engineerId} onCancel={() => setActiveForm(null)} onDone={() => setDone(true)} />
      )}
      {activeForm === "reject" && (
        <RejectForm engineerId={engineerId} onCancel={() => setActiveForm(null)} onDone={() => setDone(true)} />
      )}
    </div>
  );
}

/*
 * ApproveForm
 * Score input (0-100) + Send button → POST /api/admin/verification/[id]/review with decision=APPROVED.
 * @param engineerId - ID of engineer being approved
 * @param onCancel   - Called when admin cancels
 * @param onDone     - Called after successful approval
 */
"use client";
import { useState } from "react";
import { Button } from "@/components/ui/button";

interface Props { engineerId: string; onCancel: () => void; onDone: () => void; }

export function ApproveForm({ engineerId, onCancel, onDone }: Props) {
  const [score, setScore] = useState<string>("");
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  async function handleSubmit() {
    const scoreNum = parseInt(score, 10);
    if (isNaN(scoreNum) || scoreNum < 0 || scoreNum > 100) {
      setError("Score must be 0–100");
      return;
    }
    setLoading(true);
    setError(null);
    try {
      const res = await fetch(`/api/admin/verification/${engineerId}/review`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ decision: "APPROVED", score: scoreNum }),
      });
      const data = await res.json();
      if (!res.ok) { setError(data.error ?? "Failed"); return; }
      onDone();
    } catch { setError("Network error"); }
    finally { setLoading(false); }
  }

  return (
    <div className="space-y-3">
      <label className="block text-sm font-medium text-brand-foreground">Hiretivo Score (0–100)</label>
      <input
        type="number" min="0" max="100" value={score}
        onChange={(e) => setScore(e.target.value)}
        placeholder="e.g. 72"
        className="w-full rounded-lg border border-brand-border focus:ring-2 focus:ring-[#7560ff]/20 focus:border-[#7560ff] px-3 h-10 text-sm outline-none"
      />
      {error && <p className="text-brand-danger text-xs">{error}</p>}
      <div className="flex gap-2">
        <Button variant="secondary" onClick={onCancel} disabled={loading} size="sm">Cancel</Button>
        <Button onClick={handleSubmit} disabled={loading || !score} className="bg-brand-primary text-white hover:bg-[#5a47d9]" size="sm">
          {loading ? "Sending…" : "Send Approval"}
        </Button>
      </div>
    </div>
  );
}

/*
 * RejectForm
 * Multi-select rejection reasons + optional note + reapply window selector.
 * POSTs to /api/admin/verification/[id]/review with decision=REJECTED.
 * @param engineerId - ID of engineer being rejected
 * @param onCancel   - Called when admin cancels
 * @param onDone     - Called after successful rejection
 */
"use client";
import { useState } from "react";
import { Button } from "@/components/ui/button";
import { REJECTION_REASONS } from "@/constants/verification-reasons";

interface Props { engineerId: string; onCancel: () => void; onDone: () => void; }

export function RejectForm({ engineerId, onCancel, onDone }: Props) {
  const [selectedReasons, setSelectedReasons] = useState<string[]>([]);
  const [customNote, setCustomNote] = useState("");
  const [reapplyDays, setReapplyDays] = useState<30 | 60 | 90>(30);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  function toggleReason(key: string) {
    setSelectedReasons((prev) =>
      prev.includes(key) ? prev.filter((r) => r !== key) : [...prev, key]
    );
  }

  async function handleSubmit() {
    if (selectedReasons.length === 0) { setError("Select at least one reason"); return; }
    setLoading(true);
    setError(null);
    try {
      const res = await fetch(`/api/admin/verification/${engineerId}/review`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          decision: "REJECTED",
          reasons: selectedReasons,
          customNote: customNote.trim() || undefined,
          reapplyDays,
        }),
      });
      const data = await res.json();
      if (!res.ok) { setError(data.error ?? "Failed"); return; }
      onDone();
    } catch { setError("Network error"); }
    finally { setLoading(false); }
  }

  return (
    <div className="space-y-4">
      <div>
        <p className="text-sm font-medium text-brand-foreground mb-2">Rejection Reasons</p>
        <div className="space-y-1 max-h-56 overflow-y-auto">
          {REJECTION_REASONS.map((r) => (
            <label key={r.key} className="flex items-start gap-2 cursor-pointer hover:bg-brand-surface rounded p-1">
              <input type="checkbox" checked={selectedReasons.includes(r.key)}
                onChange={() => toggleReason(r.key)}
                className="mt-0.5 accent-brand-primary" />
              <span className="text-sm text-brand-foreground">{r.label}</span>
            </label>
          ))}
        </div>
      </div>

      <div>
        <label className="block text-sm font-medium text-brand-foreground mb-1">Custom Note (optional)</label>
        <textarea value={customNote} onChange={(e) => setCustomNote(e.target.value)} rows={3}
          placeholder="Any specific feedback for this engineer..."
          className="w-full rounded-lg border border-brand-border focus:ring-2 focus:ring-[#7560ff]/20 focus:border-[#7560ff] px-3 py-2 text-sm outline-none resize-none" />
      </div>

      <div>
        <label className="block text-sm font-medium text-brand-foreground mb-1">Reapply Window</label>
        <div className="flex gap-2">
          {([30, 60, 90] as const).map((days) => (
            <button key={days} onClick={() => setReapplyDays(days)}
              className={`flex-1 text-sm py-1.5 rounded-lg border transition-colors font-medium ${
                reapplyDays === days
                  ? "bg-brand-primary text-white border-brand-primary"
                  : "bg-white text-brand-foreground border-brand-border hover:bg-brand-surface"
              }`}>
              {days} days
            </button>
          ))}
        </div>
      </div>

      {error && <p className="text-brand-danger text-xs">{error}</p>}
      <div className="flex gap-2">
        <Button variant="secondary" onClick={onCancel} disabled={loading} size="sm">Cancel</Button>
        <Button onClick={handleSubmit} disabled={loading || selectedReasons.length === 0}
          className="bg-brand-danger text-white hover:bg-red-600" size="sm">
          {loading ? "Sending…" : "Send Rejection"}
        </Button>
      </div>
    </div>
  );
}

/*
 * ApplyButton
 * Renders the correct CTA based on the viewer's verificationStatus.
 * Used on job cards and job detail pages.
 * PIVS-10: VERIFIED → Apply, NOT_SUBMITTED/REJECTED → Get Verified to Apply →,
 *          PENDING/IN_REVIEW → disabled "Verification in progress",
 *          RECRUITER / unauthenticated → standard Apply behavior unchanged.
 * @param jobId              - Target job ID
 * @param verificationStatus - Viewer's VerificationStatus (null = unauthenticated)
 * @param userRole           - Viewer's role (null = unauthenticated)
 * @param onApply            - Callback for authenticated VERIFIED apply (opens apply modal/flow)
 * @param onLoginRequired    - Callback for unauthenticated viewer (opens LoginModal)
 */
"use client";
import { VerificationStatus, UserRole } from "@/constants/enums";
import { Button } from "@/components/ui/button";

interface Props {
  jobId: string;
  verificationStatus: VerificationStatus | null;
  userRole: UserRole | null;
  onApply: () => void;
  onLoginRequired: () => void;
}

export function ApplyButton({ jobId, verificationStatus, userRole, onApply, onLoginRequired }: Props) {
  // Unauthenticated: show Apply → triggers login modal
  if (!verificationStatus || !userRole) {
    return (
      <Button onClick={onLoginRequired} className="bg-brand-primary text-white hover:bg-[#5a47d9]">
        Apply
      </Button>
    );
  }

  // Recruiter: standard apply (recruiter viewing a job — no verification gate)
  if (userRole === UserRole.RECRUITER) {
    return (
      <Button onClick={onApply} className="bg-brand-primary text-white hover:bg-[#5a47d9]">
        Apply
      </Button>
    );
  }

  // SEEKER: gate on verificationStatus
  if (verificationStatus === VerificationStatus.VERIFIED) {
    return (
      <Button onClick={onApply} className="bg-brand-primary text-white hover:bg-[#5a47d9]">
        Apply
      </Button>
    );
  }

  if (
    verificationStatus === VerificationStatus.NOT_SUBMITTED ||
    verificationStatus === VerificationStatus.REJECTED
  ) {
    return (
      <a href="/seeker/verification"
        className="inline-block bg-brand-primary text-white font-medium text-sm px-5 py-2.5 rounded-lg hover:bg-[#5a47d9] transition-colors">
        Get Verified to Apply →
      </a>
    );
  }

  // PENDING or IN_REVIEW: disabled
  return (
    <Button disabled className="bg-brand-surface text-brand-foreground-muted border border-brand-border cursor-not-allowed">
      Verification in progress — check back soon
    </Button>
  );
}
