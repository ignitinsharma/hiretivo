/*
 * /seeker/verification
 * Server component: fetches current verification status.
 * Redirects to /seeker/dashboard if engineer is already PENDING/IN_REVIEW/VERIFIED.
 * Renders VerificationFlow client component with initial data.
 */
import { auth } from "@/auth";
import { redirect } from "next/navigation";
import { prisma } from "@/lib/prisma";
import { VerificationStatus, UserRole } from "@/constants/enums";
import { VerificationFlow } from "@/modules/verification/components/VerificationFlow";

export default async function VerificationPage() {
  const session = await auth();
  if (!session?.user?.id) redirect("/auth/seeker/login?callbackUrl=/seeker/verification");
  if (session.user.role !== UserRole.SEEKER) redirect("/seeker/dashboard");

  const user = await prisma.user.findUnique({
    where: { id: session.user.id },
    select: {
      verificationStatus: true,
      reapplyAfter: true,
      name: true,
      email: true,
      seekerProfile: {
        select: { githubUsername: true, salaryMinUsd: true, salaryMaxUsd: true },
      },
    },
  });

  // Already in progress or verified — redirect to dashboard
  const blockedStatuses = [VerificationStatus.PENDING, VerificationStatus.IN_REVIEW, VerificationStatus.VERIFIED];
  if (user?.verificationStatus && blockedStatuses.includes(user.verificationStatus)) {
    redirect("/seeker/dashboard");
  }

  // If REJECTED and reapplyAfter in future — still show page but VerificationFlow handles gate
  return (
    <main className="min-h-screen bg-brand-surface">
      <div className="max-w-2xl mx-auto px-4 py-12">
        <VerificationFlow
          initialStatus={user?.verificationStatus ?? VerificationStatus.NOT_SUBMITTED}
          reapplyAfter={user?.reapplyAfter?.toISOString() ?? null}
          prefillGithubUsername={user?.seekerProfile?.githubUsername ?? null}
          userName={user?.name ?? ""}
        />
      </div>
    </main>
  );
}

--- src/modules/verification/components/VerificationFlow.tsx ---
/*
 * VerificationFlow
 * Multi-step verification form container. Manages currentStep (1-4) and
 * shared form state across steps. Renders step components based on currentStep.
 * On final submit: POST /api/engineer/verification; shows success state.
 */
"use client";
import { useState } from "react";
import { isAfter } from "date-fns";
import { VerificationStatus } from "@/constants/enums";
import { StepGitHub } from "./steps/StepGitHub";
import { StepBioProjects } from "./steps/StepBioProjects";
import { StepVideo } from "./steps/StepVideo";
import { StepSlugSubmit } from "./steps/StepSlugSubmit";

interface Props {
  initialStatus: VerificationStatus;
  reapplyAfter: string | null;
  prefillGithubUsername: string | null;
  userName: string;
}

export interface VerificationFormData {
  githubUrl: string;
  legalName: string;
  bio: string;
  topProjects: { name: string; description: string; url: string }[];
  videoIntroUrl: string;
  profileSlug: string;
}

/*
 * Step progress indicator — 4 steps rendered as numbered circles.
 */
function StepIndicator({ current, total }: { current: number; total: number }) {
  return (
    <div className="flex items-center gap-2 mb-8">
      {Array.from({ length: total }, (_, i) => i + 1).map((step) => (
        <div key={step} className="flex items-center gap-2">
          <div className={`w-8 h-8 rounded-full flex items-center justify-center text-sm font-medium border ${
            step === current ? "bg-brand-primary text-white border-brand-primary" :
            step < current  ? "bg-brand-success text-white border-brand-success" :
                              "bg-white text-brand-foreground-muted border-brand-border"
          }`}>{step}</div>
          {step < total && <div className={`h-px w-8 ${step < current ? "bg-brand-success" : "bg-brand-border"}`} />}
        </div>
      ))}
    </div>
  );
}

export function VerificationFlow({ initialStatus, reapplyAfter, prefillGithubUsername, userName }: Props) {
  const [currentStep, setCurrentStep] = useState(1);
  const [submitted, setSubmitted] = useState(false);
  const [submitting, setSubmitting] = useState(false);
  const [submitError, setSubmitError] = useState<string | null>(null);

  const [formData, setFormData] = useState<VerificationFormData>({
    githubUrl: prefillGithubUsername ? `https://github.com/${prefillGithubUsername}` : "",
    legalName: userName,
    bio: "",
    topProjects: [{ name: "", description: "", url: "" }],
    videoIntroUrl: "",
    profileSlug: "",
  });

  // Reapply gate: REJECTED but reapplyAfter not yet passed
  const cannotReapply = initialStatus === VerificationStatus.REJECTED
    && reapplyAfter
    && isAfter(new Date(reapplyAfter), new Date());

  if (cannotReapply) {
    return (
      <div className="rounded-xl border border-brand-border bg-white p-6 text-center">
        <p className="text-brand-foreground font-semibold mb-2">Reapply window not yet open</p>
        <p className="text-brand-foreground-muted text-sm">You can resubmit after {new Date(reapplyAfter!).toLocaleDateString()}.</p>
      </div>
    );
  }

  if (submitted) {
    return (
      <div className="rounded-xl border border-brand-border bg-white p-8 text-center">
        <div className="text-4xl mb-4">⏳</div>
        <h2 className="font-bold text-xl text-brand-foreground mb-2">Under Review</h2>
        <p className="text-brand-foreground-muted">We'll email you within 48 hours with our decision.</p>
        <p className="text-brand-foreground-secondary text-sm mt-4">Your profile was submitted for human review. AI assistance may be used to help evaluate your GitHub profile.</p>
      </div>
    );
  }

  async function handleSubmit() {
    setSubmitting(true);
    setSubmitError(null);
    try {
      const res = await fetch("/api/engineer/verification", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(formData),
      });
      const data = await res.json();
      if (!res.ok) { setSubmitError(data.error ?? "Submission failed. Please try again."); return; }
      setSubmitted(true);
    } catch {
      setSubmitError("Network error. Please try again.");
    } finally {
      setSubmitting(false);
    }
  }

  return (
    <div>
      <h1 className="font-bold text-2xl md:text-3xl text-brand-foreground mb-2 tracking-tight">Get Hiretivo Verified</h1>
      <p className="text-brand-foreground-muted mb-8">Complete all steps to submit for verification. Our team reviews submissions within 48 hours.</p>
      <StepIndicator current={currentStep} total={4} />

      {currentStep === 1 && (
        <StepGitHub
          value={formData.githubUrl}
          onChange={(url) => setFormData((p) => ({ ...p, githubUrl: url }))}
          onNext={() => setCurrentStep(2)}
        />
      )}
      {currentStep === 2 && (
        <StepBioProjects
          legalName={formData.legalName}
          bio={formData.bio}
          topProjects={formData.topProjects}
          onChangeName={(v) => setFormData((p) => ({ ...p, legalName: v }))}
          onChangeBio={(v) => setFormData((p) => ({ ...p, bio: v }))}
          onChangeProjects={(v) => setFormData((p) => ({ ...p, topProjects: v }))}
          onNext={() => setCurrentStep(3)}
          onBack={() => setCurrentStep(1)}
        />
      )}
      {currentStep === 3 && (
        <StepVideo
          value={formData.videoIntroUrl}
          onChange={(v) => setFormData((p) => ({ ...p, videoIntroUrl: v }))}
          onNext={() => setCurrentStep(4)}
          onBack={() => setCurrentStep(2)}
        />
      )}
      {currentStep === 4 && (
        <StepSlugSubmit
          profileSlug={formData.profileSlug}
          formData={formData}
          onChangeSlug={(v) => setFormData((p) => ({ ...p, profileSlug: v }))}
          onSubmit={handleSubmit}
          onBack={() => setCurrentStep(3)}
          submitting={submitting}
          submitError={submitError}
        />
      )}
    </div>
  );
}

--- src/modules/verification/components/steps/StepGitHub.tsx ---
/*
 * StepGitHub
 * Step 1: engineer enters GitHub profile URL.
 * On Next: calls /api/engineer/github-scan to validate and count public repos.
 * States: IDLE | INVALID_URL | SCANNING | SUCCESS | PRIVATE | NOT_FOUND | TIMEOUT
 * PRIVATE is a warning (can proceed); NOT_FOUND blocks Next.
 */
"use client";
import { useState } from "react";
import { Button } from "@/components/ui/button";

type ScanStatus = "idle" | "invalid_url" | "scanning" | "success" | "private" | "not_found" | "timeout";

interface Props {
  value: string;
  onChange: (url: string) => void;
  onNext: () => void;
}

const GITHUB_REGEX = /^https:\/\/github\.com\/[a-zA-Z0-9]([a-zA-Z0-9-]*[a-zA-Z0-9])?$/;

export function StepGitHub({ value, onChange, onNext }: Props) {
  const [scanStatus, setScanStatus] = useState<ScanStatus>("idle");
  const [publicRepoCount, setPublicRepoCount] = useState<number | null>(null);

  async function handleNext() {
    const trimmed = value.trim();
    if (!GITHUB_REGEX.test(trimmed)) { setScanStatus("invalid_url"); return; }
    setScanStatus("scanning");
    try {
      const res = await fetch("/api/engineer/github-scan", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ githubUrl: trimmed }),
      });
      const data = await res.json() as { status: string; publicRepoCount?: number };
      if (data.status === "not_found") { setScanStatus("not_found"); return; }
      if (data.status === "private")   { setScanStatus("private"); setPublicRepoCount(0); return; }
      if (data.status === "timeout")   { setScanStatus("timeout"); return; }
      setScanStatus("success");
      setPublicRepoCount(data.publicRepoCount ?? 0);
    } catch { setScanStatus("timeout"); }
  }

  const canProceed = scanStatus === "success" || scanStatus === "private" || scanStatus === "timeout";

  return (
    <div className="rounded-xl border border-brand-border bg-white p-6">
      <h2 className="font-semibold text-xl text-brand-foreground mb-1 tracking-tight">GitHub Profile</h2>
      <p className="text-brand-foreground-muted text-sm mb-6">We'll check your public repos to evaluate your code quality.</p>

      <label className="block text-sm font-medium text-brand-foreground mb-1">GitHub URL</label>
      <input
        type="url"
        value={value}
        onChange={(e) => { onChange(e.target.value); setScanStatus("idle"); }}
        placeholder="https://github.com/username"
        className="w-full rounded-lg border border-brand-border focus:ring-2 focus:ring-[#7560ff]/20 focus:border-[#7560ff] px-3 h-10 text-sm outline-none mb-3"
      />

      {scanStatus === "invalid_url" && <p className="text-brand-danger text-sm mb-3">Please enter a valid GitHub profile URL (https://github.com/username)</p>}
      {scanStatus === "scanning"    && <p className="text-brand-foreground-muted text-sm mb-3 flex items-center gap-2"><span className="animate-spin inline-block">⟳</span> Checking GitHub profile…</p>}
      {scanStatus === "success"     && <p className="text-brand-success text-sm mb-3">✓ Found {publicRepoCount} public repositories</p>}
      {scanStatus === "private"     && <p className="text-yellow-600 text-sm mb-3">⚠ No public repos found. You can still proceed — our admin will review your profile manually.</p>}
      {scanStatus === "not_found"   && <p className="text-brand-danger text-sm mb-3">GitHub profile not found. Please check the URL and try again.</p>}
      {scanStatus === "timeout"     && <p className="text-brand-foreground-muted text-sm mb-3">GitHub scan timed out. You can proceed anyway — our team will check manually.</p>}

      <div className="flex gap-3 mt-4">
        {!canProceed ? (
          <Button
            onClick={handleNext}
            disabled={scanStatus === "scanning" || !value.trim()}
            className="bg-brand-primary text-white hover:bg-[#5a47d9]"
          >
            {scanStatus === "scanning" ? "Checking…" : "Check & Continue →"}
          </Button>
        ) : (
          <Button onClick={onNext} className="bg-brand-primary text-white hover:bg-[#5a47d9]">
            Continue →
          </Button>
        )}
      </div>
    </div>
  );
}

/*
 * StepBioProjects
 * Step 2: legal full name, bio (min 100 words, word count shown), top 3 projects.
 * Each project: name (required), description (required), URL (https only, validated).
 * Add Project button shown when fewer than 3 projects.
 */
"use client";
import { useState } from "react";
import { Button } from "@/components/ui/button";

interface Project { name: string; description: string; url: string; }
interface Props {
  legalName: string; bio: string; topProjects: Project[];
  onChangeName: (v: string) => void;
  onChangeBio:  (v: string) => void;
  onChangeProjects: (v: Project[]) => void;
  onNext: () => void; onBack: () => void;
}

function wordCount(text: string): number {
  return text.trim().split(/\s+/).filter(Boolean).length;
}

export function StepBioProjects({ legalName, bio, topProjects, onChangeName, onChangeBio, onChangeProjects, onNext, onBack }: Props) {
  const [errors, setErrors] = useState<Record<string, string>>({});

  function validate(): boolean {
    const errs: Record<string, string> = {};
    if (!legalName.trim()) errs.legalName = "Legal name is required";
    if (wordCount(bio) < 100) errs.bio = `Bio must be at least 100 words (currently ${wordCount(bio)})`;
    topProjects.forEach((p, i) => {
      if (!p.name.trim()) errs[`proj_name_${i}`] = "Project name required";
      if (!p.description.trim()) errs[`proj_desc_${i}`] = "Description required";
      if (p.url && !p.url.startsWith("https://")) errs[`proj_url_${i}`] = "URL must start with https://";
    });
    setErrors(errs);
    return Object.keys(errs).length === 0;
  }

  function updateProject(index: number, field: keyof Project, value: string) {
    const updated = [...topProjects];
    updated[index] = { ...updated[index], [field]: value };
    onChangeProjects(updated);
  }

  function addProject() {
    if (topProjects.length < 3) onChangeProjects([...topProjects, { name: "", description: "", url: "" }]);
  }

  function removeProject(index: number) {
    onChangeProjects(topProjects.filter((_, i) => i !== index));
  }

  const wc = wordCount(bio);

  return (
    <div className="rounded-xl border border-brand-border bg-white p-6 space-y-6">
      <div>
        <h2 className="font-semibold text-xl text-brand-foreground mb-1 tracking-tight">Bio & Projects</h2>
        <p className="text-brand-foreground-muted text-sm">Help our reviewers understand who you are and what you've built.</p>
      </div>

      <div>
        <label className="block text-sm font-medium text-brand-foreground mb-1">Legal Full Name</label>
        <input value={legalName} onChange={(e) => onChangeName(e.target.value)}
          className="w-full rounded-lg border border-brand-border focus:ring-2 focus:ring-[#7560ff]/20 focus:border-[#7560ff] px-3 h-10 text-sm outline-none" />
        {errors.legalName && <p className="text-brand-danger text-xs mt-1">{errors.legalName}</p>}
      </div>

      <div>
        <div className="flex justify-between items-center mb-1">
          <label className="text-sm font-medium text-brand-foreground">Bio</label>
          <span className={`text-xs ${wc >= 100 ? "text-brand-success" : "text-brand-foreground-muted"}`}>{wc} / 100 words minimum</span>
        </div>
        <textarea value={bio} onChange={(e) => onChangeBio(e.target.value)} rows={6}
          placeholder="Tell us about your engineering background, what you build, and what you're looking for..."
          className="w-full rounded-lg border border-brand-border focus:ring-2 focus:ring-[#7560ff]/20 focus:border-[#7560ff] px-3 py-2 text-sm outline-none resize-none" />
        {errors.bio && <p className="text-brand-danger text-xs mt-1">{errors.bio}</p>}
      </div>

      <div>
        <div className="flex justify-between items-center mb-3">
          <label className="text-sm font-medium text-brand-foreground">Top Projects</label>
          {topProjects.length < 3 && (
            <button onClick={addProject} className="text-xs text-brand-primary hover:underline font-medium">+ Add project</button>
          )}
        </div>
        <div className="space-y-4">
          {topProjects.map((p, i) => (
            <div key={i} className="rounded-lg border border-brand-border p-4 bg-brand-surface">
              <div className="flex justify-between items-center mb-3">
                <span className="text-xs font-medium text-brand-foreground-muted uppercase tracking-wider">Project {i + 1}</span>
                {topProjects.length > 1 && (
                  <button onClick={() => removeProject(i)} className="text-xs text-brand-danger hover:underline">Remove</button>
                )}
              </div>
              <div className="space-y-2">
                <input value={p.name} onChange={(e) => updateProject(i, "name", e.target.value)}
                  placeholder="Project name" className="w-full rounded-lg border border-brand-border focus:ring-2 focus:ring-[#7560ff]/20 focus:border-[#7560ff] px-3 h-9 text-sm outline-none" />
                {errors[`proj_name_${i}`] && <p className="text-brand-danger text-xs">{errors[`proj_name_${i}`]}</p>}
                <textarea value={p.description} onChange={(e) => updateProject(i, "description", e.target.value)}

const SLUG_REGEX = /^[a-z0-9][a-z0-9-]*[a-z0-9]$/;
