    Full implementation:

    ```typescript
    import { NextRequest, NextResponse } from "next/server";
    import bcrypt from "bcryptjs";
    import { Role } from "@prisma/client";
    import { prisma } from "@/lib/prisma";
    import { registerSchema } from "@/lib/validations/auth";

    export async function POST(req: NextRequest) {
      try {
        const body = await req.json();

        // Accept optional companyName (passed by recruiter register form but not in registerSchema)
        const parsed = registerSchema.safeParse(body);
        if (!parsed.success) {
          return NextResponse.json(
            { error: "Invalid input", details: parsed.error.flatten().fieldErrors },
            { status: 400 }
          );
        }

        const { email, password, name, role } = parsed.data;
        const normalizedEmail = email.toLowerCase();

        const existing = await prisma.user.findUnique({
          where: { email: normalizedEmail },
          select: { id: true },
        });

        if (existing) {
          return NextResponse.json(
            { error: "An account with this email already exists." },
            { status: 409 }
          );
        }

        const passwordHash = await bcrypt.hash(password, 12);

        const user = await prisma.user.create({
          data: {
            email: normalizedEmail,
            name: name ?? null,
            passwordHash,
            role: role as Role,
          },
          select: { id: true, email: true, role: true },
        });

        return NextResponse.json({ id: user.id, email: user.email, role: user.role }, { status: 201 });
      } catch (err) {
        console.error("[register] unexpected error:", err);
        return NextResponse.json({ error: "Internal server error" }, { status: 500 });
      }
    }
    ```

    Key decisions (Claude's discretion per CONTEXT.md):
    - Salt rounds: 12 (good balance of security vs latency on Vercel Edge)
    - Error shape: `{ error: string }` for client-parseable messages
    - 409 for duplicate email (not 400)
    - companyName not stored here — recruiter onboarding server action handles RecruiterProfile creation (Plan 05)
    - Does NOT send verification email — that is Plan 03's responsibility; Plan 02 only creates the User

    ```typescript
    const onSubmit = async (data: SeekerRegisterInput) => {
      setIsSubmitting(true);
      try {
        // Step 1: Create user account (per D-01)
        const registerRes = await fetch("/api/auth/register", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({
            email: data.email.toLowerCase(),
            password: data.password,
            name: data.name,
            role: "SEEKER",
          }),
        });

        if (!registerRes.ok) {
          const err = await registerRes.json();
          if (registerRes.status === 409) {
            toast.error("An account with this email already exists.");
          } else {
            toast.error(err.error ?? "Failed to create account. Please try again.");
          }
          return;
        }

        // Step 2: Sign in immediately after creation (per D-01)
        const result = await signIn("credentials", {
          email: data.email.toLowerCase(),
          password: data.password,
          redirect: false,
        });

        if (result?.error) {
          toast.error("Account created but sign-in failed. Please try logging in.");
          return;
        }

        toast.success("Account created successfully!");
        router.push("/seeker/onboarding");
        router.refresh();
      } catch {
        toast.error("Something went wrong. Please try again.");
      } finally {
        setIsSubmitting(false);
      }
    };
    ```
