    Create `src/app/actions/onboarding/recruiter.ts`. The `src/app/actions/onboarding/` directory was created in Plan 04.

    Full implementation:

    ```typescript
    "use server";

    import { auth } from "@/auth";
    import { prisma } from "@/lib/prisma";

    type ActionResult = { success: true } | { success: false; error: string };

    export async function saveRecruiterProfile(data: {
      companyName: string;
      website: string | null;
      companySize: string | null;
      bio?: string | null;
    }): Promise<ActionResult> {
      const session = await auth();
      if (!session?.user?.id) {
        return { success: false, error: "Not authenticated" };
      }

      if (!data.companyName || data.companyName.trim().length === 0) {
        return { success: false, error: "Company name is required" };
      }

      await prisma.recruiterProfile.upsert({
        where: { userId: session.user.id },
        create: {
          userId: session.user.id,
          companyName: data.companyName.trim(),
          website: data.website ?? null,
          companySize: data.companySize ?? null,
          bio: data.bio ?? null,
        },
        update: {
          companyName: data.companyName.trim(),
          website: data.website ?? null,
          companySize: data.companySize ?? null,
          bio: data.bio ?? null,
        },
      });

      return { success: true };
    }
    ```

    Notes:
    - upsert means re-submitting the form overwrites previous data (safe for onboarding)
    - website maps to companyWebsite from the store (transform in the calling component)
    - bio is optional — the recruiter store's step2 does not have a bio field, so pass null
    - companySize is a string (e.g. "11-50") matching the CompanySize type values
