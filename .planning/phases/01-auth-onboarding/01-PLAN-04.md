    ```typescript
    "use server";

    import { auth } from "@/auth";
    import { prisma } from "@/lib/prisma";

    type ActionResult = { success: true } | { success: false; error: string };

    // --- Step 1: Situation ---
    // experienceLevel is not a DB column on SeekerProfile; situation is the only Step 1 field
    // persisted to DB. Per D-05, the schema has: situation, primaryRole, salaryMin, salaryMax,
    // currency, availability. experienceLevel informs the UI wizard but is not stored separately.
    export async function saveStep1(data: {
      situation: string;
      experienceLevel: string;
    }): Promise<ActionResult> {
      const session = await auth();
      if (!session?.user?.id) return { success: false, error: "Not authenticated" };

      await prisma.seekerProfile.upsert({
        where: { userId: session.user.id },
        create: {
          userId: session.user.id,
          situation: data.situation,
        },
        update: {
          situation: data.situation,
        },
      });

      return { success: true };
    }

    // --- Step 2: Primary Role + Years + Skills ---
    // Skills are stored as a JSON-stringified array in the headline column as a temporary
    // measure. Phase 2/3 will migrate to the SeekerSkill relation when skill seeds are available.
    export async function saveStep2(data: {
      primaryRole: string;
      yearsOfExperience: string | null;
      skills: string[];
    }): Promise<ActionResult> {
      const session = await auth();
      if (!session?.user?.id) return { success: false, error: "Not authenticated" };

      await prisma.seekerProfile.upsert({
        where: { userId: session.user.id },
        create: {
          userId: session.user.id,
          primaryRole: data.primaryRole,
          headline: JSON.stringify(data.skills),
        },
        update: {
          primaryRole: data.primaryRole,
          headline: JSON.stringify(data.skills),
        },
      });

      return { success: true };
    }

    // --- Step 3: Resume (file already uploaded to R2 by UI) ---
    // resumeUrl stored in bio field temporarily until a dedicated resumeUrl column is added.
    // Step 3 is skippable so null values are valid.
    export async function saveStep3(data: {
      resumeKey: string | null;
      resumeUrl: string | null;
    }): Promise<ActionResult> {
      const session = await auth();
      if (!session?.user?.id) return { success: false, error: "Not authenticated" };

      await prisma.seekerProfile.upsert({
        where: { userId: session.user.id },
        create: {
          userId: session.user.id,
          bio: data.resumeUrl,
        },
        update: {
          bio: data.resumeUrl,
        },
      });

      return { success: true };
    }

    // --- Step 4: Job Preferences ---
    export async function saveStep4(data: {
      salaryMin: number | null;
      salaryMax: number | null;
      currency: string;
      availability: string | null;
      workPreference: string | null;
    }): Promise<ActionResult> {
      const session = await auth();
      if (!session?.user?.id) return { success: false, error: "Not authenticated" };

      await prisma.seekerProfile.upsert({
        where: { userId: session.user.id },
        create: {
          userId: session.user.id,
          salaryMin: data.salaryMin,
          salaryMax: data.salaryMax,
          currency: data.currency,
          availability: data.availability,
          workPreference: data.workPreference as any,
        },
        update: {
          salaryMin: data.salaryMin,
          salaryMax: data.salaryMax,
          currency: data.currency,
          availability: data.availability,
          workPreference: data.workPreference as any,
        },
      });

      return { success: true };
    }

    // --- Profile Completeness (per D-09: computed on read, not stored) ---
    export async function computeProfileCompleteness(userId: string): Promise<number> {
      const profile = await prisma.seekerProfile.findUnique({
        where: { userId },
        select: {
          situation: true,
          primaryRole: true,
          headline: true,    // skills stored here temporarily
          bio: true,         // resume URL stored here temporarily
          salaryMin: true,
          salaryMax: true,
          availability: true,
        },
      });

      if (!profile) return 0;

      // 7 tracked fields across steps 1–4
      const fields = [
        profile.situation,
        profile.primaryRole,
        profile.headline,    // skills (step 2)
        profile.bio,         // resume URL (step 3)
        profile.salaryMin,
        profile.salaryMax,
        profile.availability,
      ];

      const filled = fields.filter((f) => f !== null && f !== undefined && f !== "").length;
      return Math.round((filled / fields.length) * 100);
    }
    ```
